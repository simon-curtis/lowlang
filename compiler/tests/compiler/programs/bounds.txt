fn sum(xs: []i64) -> i64 {
    var t = 0
    for i in 0..xs.len {
        t += xs[i]
    }
    t
}

fn nested(xs: []i64) -> i64 {
    var t = 0
    for i in 0..xs.len {
        for j in 0..xs.len {
            t += xs[i] * xs[j]
        }
    }
    t
}

fn bump(xs: []i64, i: i64) {
    xs[i] += 1
    xs[i] += 1
}

fn grow() -> i64 {
    var xs: []i64 = [1]
    var t = 0
    var i = 0
    for i < xs.len {
        push(mut xs, i)
        t += xs[i]
        i += 1
        if i > 5 {
            return t
        }
    }
    t
}

fn main() {
    let xs = [1, 2, 3]
    print(sum(xs))
    print(nested(xs))
    bump(xs, 1)
    print(grow())
}
