fn spread(xs: []i64, scale: i64) -> i64 {
    var total = 0
    for i in 0..xs.len {
        total += xs[i] * scale
    }
    for i in 0..xs.len {
        total += xs[i]
    }
    total
}

fn main() {
    var xs: []i64 = []
    for i in 0..10 {
        push(mut xs, i)
    }
    print(spread(xs, 3))
}
