fn helper(x: i64) -> i64 {
    x * 3 + 1
}

fn small(a: i64, k: i64) -> i64 {
    var r = a
    if k > 0 {
        r = helper(r)
    }
    r + 7
}

fn large(a: i64, b: i64, k: i64) -> i64 {
    var r = a
    if k > 0 {
        r = helper(r)
    }
    let p = r * 3 + b
    let q = (p ^ r) + (b << 2)
    let s = (q - p) * (r + 5)
    let t = (s & 0xffff) + (q | 3) - (p >> 1)
    let u = t * 7 + s - q
    let v = (u ^ t) + (u & s) + (r | b)
    v - u + t * 2
}

fn main() {
    print(small(1, 1) + large(1, 2, 1))
}
