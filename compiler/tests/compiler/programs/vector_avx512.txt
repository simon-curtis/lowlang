import std/simd

fn multiply64(a: i64x4, b: i64x4) -> i64x4 {
    a * b
}

fn minimum64(a: i64x4, b: i64x4) -> i64x4 {
    simd.min(a, b)
}

fn maximum_unsigned64(a: u64x2, b: u64x2) -> u64x2 {
    simd.max(a, b)
}

fn shift_signed64(a: i64x4) -> i64x4 {
    a >> 3
}

fn less_unsigned8(a: u8x32, b: u8x32) -> u8x32 {
    a < b
}

fn not_equal32(a: i32x8, b: i32x8) -> i32x8 {
    a != b
}

fn less_equal64(a: i64x2, b: i64x2) -> i64x2 {
    a <= b
}

fn select32(m: i32x8, a: i32x8, b: i32x8) -> i32x8 {
    simd.blend(m, a, b)
}

fn invert16(a: u16x16) -> u16x16 {
    ~a
}

fn broadcast16(x: u16) -> u16x16 {
    simd.splat(x)
}

fn broadcast64(x: u64) -> u64x4 {
    simd.splat(x)
}

fn mask16(a: u16x16) -> u64 {
    simd.mask(a)
}

fn mask16_narrow(a: u16x8) -> u64 {
    simd.mask(a)
}

fn add32(a: i32x8, b: i32x8) -> i32x8 {
    a + b
}

fn multiply_bytes(a: u8x32, b: u8x32) -> u8x32 {
    a * b
}

fn shift_bytes(a: u8x16, n: i64) -> u8x16 {
    a << n
}

fn mask_bytes(a: u8x32) -> u64 {
    simd.mask(a)
}

fn mask_quads(a: f64x2) -> u64 {
    simd.mask(a)
}

fn any_equal(a: u8x32, b: u8x32) -> bool {
    simd.any(a == b)
}

fn all_below(a: u8x32, b: u8x32) -> bool {
    simd.all(a < b)
}

fn mask_and_keep(a: u32x8, b: u32x8) -> u64 {
    let m = a < b
    let bits = simd.mask(m)
    let picked = simd.blend(m, a, b)
    bits + simd.mask(picked)
}

fn add512(a: u32x16, b: u32x16) -> u32x16 {
    a + b
}

fn and512(a: u64x8, b: u64x8) -> u64x8 {
    a & b
}

fn equal512(a: i32x16, b: i32x16) -> i32x16 {
    a == b
}

fn less_float512(a: f32x16, b: f32x16) -> f32x16 {
    a < b
}

fn negate_float512(a: f32x16) -> f32x16 {
    -a
}

fn invert512(a: u16x32) -> u16x32 {
    ~a
}

fn multiply_bytes512(a: u8x64, b: u8x64) -> u8x64 {
    a * b
}

fn shift_bytes512(a: u8x64) -> u8x64 {
    a >> 2
}

fn mask_bytes512(a: u8x64) -> u64 {
    simd.mask(a)
}

fn mask_words512(a: u16x32) -> u64 {
    simd.mask(a)
}

fn main() {
    let a: i64x4 = simd.splat(5)
    let b: i64x4 = simd.splat(3)
    let c: i64x2 = simd.splat(4)
    let u: u64x2 = simd.splat(9)
    let bytes: u8x32 = simd.splat(7)
    let words: u16x16 = simd.splat(2)
    let narrow: u16x8 = simd.splat(2)
    let d: i32x8 = simd.splat(1)
    print(multiply64(a, b))
    print(minimum64(a, b))
    print(maximum_unsigned64(u, u))
    print(shift_signed64(a))
    print(less_unsigned8(bytes, bytes))
    print(not_equal32(d, d))
    print(less_equal64(c, c))
    print(select32(d, d, d))
    print(invert16(words))
    print(broadcast16(4))
    print(broadcast64(6))
    print(mask16(words))
    print(mask16_narrow(narrow))
    print(add32(d, d))
    let wide_bytes: u8x64 = simd.splat(3)
    let wide_words: u16x32 = simd.splat(1)
    let wide_ints: i32x16 = simd.splat(2)
    let wide_floats: f32x16 = simd.splat(1.5)
    let wide_quads: u64x8 = simd.splat(12)
    let floats: f64x2 = simd.splat(-1.0)
    print(multiply_bytes(bytes, bytes))
    print(shift_bytes(simd.splat(9), 2))
    print(mask_bytes(bytes))
    print(mask_quads(floats))
    print(any_equal(bytes, bytes))
    print(all_below(bytes, bytes))
    print(mask_and_keep(simd.splat(1), simd.splat(2)))
    print(add512(simd.splat(4), simd.splat(5)))
    print(and512(wide_quads, wide_quads))
    print(equal512(wide_ints, wide_ints))
    print(less_float512(wide_floats, wide_floats))
    print(negate_float512(wide_floats))
    print(invert512(wide_words))
    print(multiply_bytes512(wide_bytes, wide_bytes))
    print(shift_bytes512(wide_bytes))
    print(mask_bytes512(wide_bytes))
    print(mask_words512(wide_words))
}
