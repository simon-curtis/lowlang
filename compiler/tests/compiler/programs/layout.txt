fn fib(n: i64) -> i64 {
    if n < 2 { n } else { fib(n - 1) + fib(n - 2) }
}

fn sieve(limit: i64) -> i64 {
    var composite: []u8 = []
    for i in 0..limit {
        push(mut composite, u8(0))
    }
    var count = 0
    for i in 2..limit {
        if composite[i] == 0 {
            count += 1
            var j = i * i
            for j < limit {
                composite[j] = 1
                j += i
            }
        }
    }
    count
}

fn search(xs: []i64, key: i64) -> i64 {
    var found = -1
    for i in 0..xs.len {
        if xs[i] == key {
            found = i
            break
        }
        if xs[i] < 0 {
            continue
        }
    }
    found
}

fn main() {
    print(fib(10))
    print(sieve(100))
    print(search([1, 2, 3], 2))
}
