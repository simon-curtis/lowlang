fn main() {
    let limit = LIMIT
    var composite: []u8 = []
    for i in 0..limit {
        push(mut composite, u8(0))
    }
    var count = 0
    for i in 2..limit {
        if composite[i] == 0 {
            count += 1
            var j = i * i
            for j < limit {
                composite[j] = 1
                j += i
            }
        }
    }
    print(count)
}
