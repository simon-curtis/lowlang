import std/text

/// A box.
pub struct Box {
    /// Across.
    pub width: i64
    secret: i64
}

pub enum Shape {
    Dot
    Square(i64)
}

pub fn area(b: *Box) -> i64 {
    return b.width
}

pub fn unit() -> Box {
    return Box { width: 1, secret: 0 }
}

fn hidden() {}

const base = 8

/// The most boxes in a row.
pub const limit = base * 4

pub const scale: f32 = 1.5

pub const clamp = u8(300)

pub const label = "box" + "es"
