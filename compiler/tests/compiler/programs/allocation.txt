
fn join(a: str, b: str, c: str, d: str, e: str) -> str {
    a + b + c + d + e
}

fn label(n: i64, name: str) -> str {
    "{name}={n}:{n}"
}

fn status(code: i64) -> str {
    let m: [i64: str] = [200: "OK", 404: "Not Found"]
    m[code] ?? "?"
}

fn names(i: i64) -> str {
    let xs = ["alpha", "beta", "gamma"]
    xs[i % xs.len]
}

fn counter() -> fn() -> i64 {
    let base = 5
    fn() -> i64 { base + 1 }
}

fn bumper() -> fn() -> i64 {
    var n = 0
    fn() -> i64 {
        n += 1
        n
    }
}

fn unused() -> i64 {
    let xs = [1, 2, 3]
    xs[0]
}

fn main() {
    print(join("a", "b", "c", "d", "e"))
    print(label(1, "x"))
    print(status(200))
    print(names(1))
    print(counter()())
    print(bumper()())
}
