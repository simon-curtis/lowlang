fn apply(xs: []i64, f: fn(i64) -> i64) -> i64 {
    var a = 0
    for x in xs {
        a += f(x)
    }
    var b = 0
    for x in xs {
        b += f(x) * 2
    }
    a + b
}

fn offset(xs: []i64, k: i64) -> i64 {
    apply(xs, fn(x: i64) -> i64 { x + k })
}

fn main() {
    let xs = [1, 2, 3]
    print(offset(xs, 1))
    print(offset(xs, 5))
}
