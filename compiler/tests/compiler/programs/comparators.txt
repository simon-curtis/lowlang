fn insertion(xs: []i64, less: fn(i64, i64) -> bool) {
    for i in 1..xs.len {
        var j = i
        for j > 0 && less(xs[j], xs[j - 1]) {
            let t = xs[j]
            xs[j] = xs[j - 1]
            xs[j - 1] = t
            j -= 1
        }
    }
}

fn main() {
    var a: []i64 = [5, 3, 9, 1, 7]
    var b: []i64 = [5, 3, 9, 1, 7]
    insertion(a, fn(x: i64, y: i64) -> bool { x < y })
    insertion(b, fn(x: i64, y: i64) -> bool { x > y })
    print(a[0])
    print(b[0])
}
