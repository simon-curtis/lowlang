import std/simd
import std/sys

fn fib(n: i64) -> i64 {
    if n < 2 {
        return n
    }
    fib(n - 1) + fib(n - 2)
}

fn across(data: u32[]) -> u32x8 {
    var acc: u32x8 = simd.splat(0)
    let cap: u32x8 = simd.splat(1000)
    var i = 0
    for i + 8 <= data.len {
        let chunk: u32x8 = simd.read(data, i)
        acc += simd.min(chunk, cap)
        let n = fib(i)
        acc += simd.splat(u32(n)) + cap
        i += 8
    }
    acc
}

fn across_syscall(data: u32[]) -> u32x8 {
    var acc: u32x8 = simd.splat(0)
    let cap: u32x8 = simd.splat(1000)
    var i = 0
    for i + 8 <= data.len {
        let chunk: u32x8 = simd.read(data, i)
        acc += simd.min(chunk, cap)
        let pid = sys.pid()
        acc += simd.splat(u32(pid)) + cap
        i += 8
    }
    acc
}

fn main() {
    print(across([1, 2, 3, 4, 5, 6, 7, 8, 9]))
    print(across_syscall([1, 2, 3, 4, 5, 6, 7, 8, 9]))
}
