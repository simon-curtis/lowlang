fn grow_in_place(n: i64) -> i64 {
    var xs: []i64 = []
    for i in 0..n {
        push(mut xs, i)
    }
    xs.len
}

fn append_two(mut xs: []i64) {
    push(mut xs, 1)
    push(mut xs, 2)
}

fn passed_on(n: i64) -> i64 {
    var xs: []i64 = []
    append_two(mut xs)
    xs.len + n
}

fn main() {
    print(grow_in_place(5))
    print(passed_on(5))
}
