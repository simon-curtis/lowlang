import std/simd

fn accumulate(data: u32[]) -> u32x8 {
    var acc: u32x8 = simd.splat(0)
    let cap: u32x8 = simd.splat(1000)
    var i = 0
    for i + 8 <= data.len {
        let chunk: u32x8 = simd.read(data, i)
        acc += simd.min(chunk, cap)
        i += 8
    }
    acc
}

fn main() {
    print(accumulate([1, 2, 3, 4, 5, 6, 7, 8, 9]))
}
