enum List {
    Cons(i64, *List)
    Nil
}

fn build(n: i64) -> List {
    if n == 0 {
        return List.Nil
    }
    Cons(n, new(build(n - 1)))
}

fn total(l: List) -> i64 {
    match l {
        Cons(v, rest) => v + total(*rest)
        Nil => 0
    }
}

fn done() -> bool {
    DONE
}
