fn value() -> i64 {
    VALUE
}

fn native(n: i64) -> i64 {
    n + VALUE
}

fn main() {
    print(0)
    let named: fn() -> i64 = value
    let inline = fn() -> i64 { VALUE }
    let callback: extern fn(i64) -> i64 = native
    var n = named() + inline() + callback(0)
    for n < 6 {
        n = named() + inline() + callback(0)
    }
    print(n)
}
