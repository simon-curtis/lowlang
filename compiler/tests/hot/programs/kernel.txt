kernel fn bump(id: u32, mut out: u32[]) {
    out[id] = id + VALUE
}

fn stamp() -> i64 {
    var out: u32[] = [0]
    var sum = 0
    for byte in launch(bump, 1, mut out).module {
        sum = (sum * 31 + i64(byte)) % 1000003
    }
    sum
}

fn main() {
    print(0)
    let first = stamp()
    var rounds = 0
    for stamp() == first {
        rounds += 1
    }
    print(rounds > 0)
}
