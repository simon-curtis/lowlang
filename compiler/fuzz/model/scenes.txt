    acc
}

fn @S_serve(ctx: *cancel.Token, jobs: chan[i64], out: chan[i64], group: sync.WaitGroup) {
    defer sync.done(group)
    let own = cancel.child(ctx)
    for {
        let stop = select {
            <-cancel.done(own) => true
            v = <-jobs => {
                out <- @S_work(v ?? 0)
                false
            }
        }
        if stop {
            break
        }
    }
}

fn @S_pause(ctx: *cancel.Token, counter: atomic.Int, group: sync.WaitGroup) {
    defer sync.done(group)
    if !cancel.pause(ctx, 60000) {
        atomic.add_int(counter, 1)
    }
}

fn @S() -> i64 {
    let ctx = cancel.token()
    let jobs = chan[i64](@C)
    let out = chan[i64](@N)
    let counter = atomic.int(0)
    let group = sync.wait_group()
    sync.add(group, 2 * @T)
    for _ in 0..@T {
        spawn @S_serve(ctx, jobs, out, group)
        spawn @S_pause(ctx, counter, group)
    }
    for i in 0..@N {
        jobs <- i
    }
    var sum = 0
    for _ in 0..@N {
        sum += (<-out) ?? 0
    }
    cancel.cancel(ctx)
    sync.wait(group)
    sum + atomic.load_int(counter)
}

fn @S_dive(n: i64, pad: @P) -> i64 {
    if n == 0 {
        return pad@F
    }
    var next = pad
    next@F += @S_work(n)
    next@L = n
    @S_dive(n - 1, next) + next@L
}

    acc: i64
    rows: [][]i64
    label: str
}

fn @S_fill(seed: i64, n: i64) -> *@UMsg {
    var rows: [][]i64 = []
    var names: [i64: str] = [:]
    var acc = 0
    for i in 0..n {
        let v = @S_work(seed + i)
        var row: []i64 = []
        for j in 0..(i % 13) + 1 {
            push(mut row, v + j)
        }
        push(mut rows, row)
        if rows.len > 16 {
            rows = rows[8..]
        }
        names[i % 29] = "{v}"
        acc += rows[0][0] ^ row[row.len - 1]
        acc += (names[(i * 7) % 29] ?? "").len
    }
    new(@UMsg { acc, rows, label: "{acc}" })
}

fn @S_send(out: chan[*@UMsg], seed: i64, n: i64) {

}

                sum += m.acc + m.rows.len + m.label.len
            }
            none => {}
        }
    }
    sum
}

fn @S_write(fd: i64, n: i64, chunk: i64) {
    var buf: []u8 = []
    for i in 0..n {
        push(mut buf, u8((@S_work(i & 255) + i) & 255))
        if buf.len == chunk {
            sys.write_all(fd, buf) or {}
            buf = []
        }
    }
    sys.write_all(fd, buf) or {}
    sys.close_fd(fd)
}

fn @S_read(fd: i64, out: chan[i64]) {
    var buf: []u8 = []
    for _ in 0..512 {
        push(mut buf, 0)
    }
    var acc = 0
    for {
        let n = sys.read(fd, buf) or break
        if n == 0 {
            break
        }
        for i in 0..n {
            acc = acc * 31 + i64(buf[i])
        }
    }
    sys.close_fd(fd)
    out <- acc
}

        sleep((x + r) % (@D + 1))
        select {
            after((x * r) % (@C + 1)) => {}

    sync.done(group)

    atomic.load_int(counter)
