fn first(xs: []str) -> str? {
    xs[0] when xs.len > 0
}

fn when(n: i64) -> i64? {
    let when = n |> when() when n > 1 && true
    let pick = fn(n: i64) n * 2 when n > 1
    return pick(when ?? 0) when when == none
}
