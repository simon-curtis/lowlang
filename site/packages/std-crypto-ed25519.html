signing

import std/crypto/ed25519
import std/text

fn unhex(s: str) -> []u8 {
    text.decode_hex(s) or []
}

fn check(label: str, public: []u8, message: []u8, signature: []u8) {
    let verdict = if ed25519.verify(public, message, signature) { "valid" } else { "rejected" }
    print("{label}: {verdict}")
}

/// Signs a message with the RFC 8032 test seed, verifies it, and shows that an altered message, an altered signature and a signature under another key are rejected.
fn main() {
    let seed = unhex("4ccd089b28ff96da9db6c346ec114e0f5b8a319f35aba624da8cf6ed4fb8a6fb")
    let public = ed25519.public_key(seed)
    print("public    {text.encode_hex(public)}")

    let message = unhex("72")
    let signature = ed25519.sign(seed, message)
    print("signature {text.encode_hex(signature)}")
    print("repeatable: {text.encode_hex(signature) == text.encode_hex(ed25519.sign(seed, message))}")

    check("original", public, message, signature)
    check("other message", public, unhex("73"), signature)

    var forged = ed25519.sign(seed, message)
    forged[10] ^= 0x01
    check("flipped bit", public, message, forged)
    check("truncated", public, message, signature[..63])

    let stranger = ed25519.public_key(ed25519.generate())
    check("other key", stranger, message, signature)
}

Functions

generate

pub fn generate() -> []u8

Returns a new 32-byte Ed25519 seed, the private key, from random in std/crypto.

public_key

pub fn public_key(seed: []u8) -> []u8

Returns the 32-byte Ed25519 public key for the 32-byte seed (RFC 8032).

The length of seed is not validated.

sign

pub fn sign(seed: []u8, message: []u8) -> []u8

Returns the 64-byte Ed25519 signature of message under the 32-byte seed (RFC 8032 section 5.1).

The signature is the 32-byte encoded point R followed by the 32-byte scalar S. Signing is deterministic: the same seed and message give the same signature. The length of seed is not validated.

verify

pub fn verify(public: []u8, message: []u8, signature: []u8) -> bool

Returns whether signature is a valid Ed25519 signature of message under the 32-byte public key (RFC 8032 section 5.1.7).

Returns false when signature is not 64 bytes, S is not below the group order, or public does not decode to a curve point. Otherwise it compares the encoding of [S]B - [k]A with R.
