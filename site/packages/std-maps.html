Imports std/slices.

Examples

layered_config

import std/maps
import std/slices

struct Request {
    path: str
    status: i64
}

/// Merges configuration layers, then summarises a request log by path and by status.
fn main() {
    let defaults: [str: str] = ["port": "8080", "log": "info", "mode": "prod"]
    let file: [str: str] = ["port": "9000", "tls": "on"]
    let flags: [str: str] = ["log": "debug"]

    let config = maps.merge(maps.merge(defaults, file), flags)
    for key in maps.sorted_keys(config) {
        print("{key} = {config[key] ?? ""}")
    }
    print("defaults untouched: {defaults["port"] ?? ""}")
    print("timeout set: {maps.has(config, "timeout")}, timeout: {maps.get_or(config, "timeout", "30s")}")

    let log = [
        Request { path: "/home", status: 200 },
        Request { path: "/login", status: 302 },
        Request { path: "/home", status: 200 },
        Request { path: "/missing", status: 404 },
        Request { path: "/home", status: 500 },
        Request { path: "/login", status: 200 }
    ]
    let per_path = maps.count_by(log, fn(r) r.path)
    print("requests per path: {per_path}")

    let by_status = maps.group_by(log, fn(r) r.status / 100)
    for class in maps.sorted_keys(by_status) {
        let paths = slices.map(by_status[class] ?? [], fn(r) r.path)
        print("{class}xx: {paths}")
    }

    let busy = maps.filter(per_path, fn(path, count) count > 1)
    print("busy paths: {maps.keys(busy)}")
    let share = maps.map_values(per_path, fn(count) count * 100 / log.len)
    print("share of traffic in percent: {share}")
}

Functions

clone

pub fn clone[K: Hash, V](m: [K: V]) -> [K: V]

Returns a new map with the same entries as m.

Changing the copy does not change m.

count_by

pub fn count_by[T, K: Hash](xs: []T, key: fn(T) -> K) -> [K: i64]

Returns how many elements of xs produce each key under key.

Keys appear in order of first occurrence.

entries

pub fn entries[K: Hash, V](m: [K: V]) -> []slices.Pair[K, V]

Returns the entries of m as pairs in insertion order, with the key in first and the value in second.

filter

pub fn filter[K: Hash, V](m: [K: V], keep: fn(K, V) -> bool) -> [K: V]

Returns a new map of the entries of m for which keep returns true, in their original order.

from_pairs

pub fn from_pairs[K: Hash, V](pairs: []slices.Pair[K, V]) -> [K: V]

Returns a map built from pairs, taking each key from first and its value from second.

When a key appears more than once, the last pair's value wins and the key keeps the position of its first appearance.

get_or

pub fn get_or[K: Hash, V](m: [K: V], key: K, fallback: V) -> V

Returns the value stored under key in m, or fallback when there is no entry.

group_by

pub fn group_by[T, K: Hash](xs: []T, key: fn(T) -> K) -> [K: []T]

Groups the elements of xs by the key that key returns for each.

Every group keeps its elements in their original order, and groups appear in order of first occurrence of their key.

has

pub fn has[K: Hash, V](m: [K: V], key: K) -> bool

Returns true when m has an entry for key.

invert

pub fn invert[K: Hash, V: Hash](m: [K: V]) -> [V: K]

Returns a new map from the values of m to their keys.

When several keys share a value, the key that comes last in insertion order is kept.

keys

pub fn keys[K: Hash, V](m: [K: V]) -> []K

Returns the keys of m in insertion order.

map_values

pub fn map_values[K: Hash, V, U](m: [K: V], f: fn(V) -> U) -> [K: U]

Returns a new map with the keys of m and f applied to each value.

merge

pub fn merge[K: Hash, V](base: [K: V], over: [K: V]) -> [K: V]

Returns a new map holding the entries of base and then those of over.

When a key is in both, the value from over wins and the key keeps its position from base. Neither argument changes.

sorted_keys

pub fn sorted_keys[K: Hash + Ord, V](m: [K: V]) -> []K

Returns the keys of m in ascending order.

values

pub fn values[K: Hash, V](m: [K: V]) -> []V

Returns the values of m in the insertion order of their keys.
