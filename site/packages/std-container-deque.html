A ring-buffer Deque[T]: push_front, push_back, pop_front, pop_back, front, back, at and size, doubling when full.

Examples

window_max

import std/container/deque

struct Reading {
    index: i64
    value: i64
}

/// Tracks the maximum of the last four sensor readings with a deque of candidates that shrinks from both ends.
fn main() {
    let readings = [3, 8, 2, 5, 9, 1, 1, 7, 4, 6]
    let width = 4
    var candidates = deque.deque[Reading]()
    var maxima: []i64 = []

    for i, value in readings {
        for {
            let last = candidates.back() ?? break
            if last.value > value {
                break
            }
            candidates.pop_back()
        }
        candidates.push_back(Reading { index: i, value })

        let oldest = candidates.front() ?? continue
        if oldest.index <= i - width {
            candidates.pop_front()
        }
        if i >= width - 1 {
            let best = candidates.front() ?? continue
            push(mut maxima, best.value)
            print("window ending at {i}: max {best.value}, {candidates.size()} candidates")
        }
    }
    print("maxima {maxima}")

    var recent = deque.deque[str]()
    recent.push_front("c")
    recent.push_front("b")
    recent.push_front("a")
    recent.push_back("d")
    print("second from the front: {recent.at(1) ?? "none"}, past the end: {recent.at(9) ?? "none"}")
}

Output

window ending at 4: max 9, 1 candidates
window ending at 5: max 9, 2 candidates
window ending at 6: max 9, 2 candidates
window ending at 7: max 9, 2 candidates
window ending at 8: max 7, 2 candidates
window ending at 9: max 7, 2 candidates
maxima [8, 9, 9, 9, 9, 7, 7]
second from the front: b, past the end: none

Types

Deque

pub struct Deque[T] {
    ...
}

A double-ended queue stored in a ring buffer.

Pushes and pops at either end take amortised constant time. The buffer doubles when it is full. Mutating functions take the deque as mut.

at

pub fn at[T](d: Deque[T], i: i64) -> T?

Returns the item i places from the front of d, where 0 is the front, or none when i is negative or at least size(d).

back

pub fn back[T](d: Deque[T]) -> T?

Returns the item at the back of d without removing it, or none when it is empty.

front

pub fn front[T](d: Deque[T]) -> T?

Returns the item at the front of d without removing it, or none when it is empty.

pop_back

pub fn pop_back[T](mut d: Deque[T]) -> T?

Removes and returns the item at the back of d, or none when it is empty.

pop_front

pub fn pop_front[T](mut d: Deque[T]) -> T?

Removes and returns the item at the front of d, or none when it is empty.

push_back

pub fn push_back[T](mut d: Deque[T], item: T)

Adds item at the back of d.

Takes amortised constant time; a full buffer is doubled and its items are copied.

push_front

pub fn push_front[T](mut d: Deque[T], item: T)

Adds item at the front of d.

Takes amortised constant time; a full buffer is doubled and its items are copied.

size

pub fn size[T](d: Deque[T]) -> i64

Returns the number of items in d.
