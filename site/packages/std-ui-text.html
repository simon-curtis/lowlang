Examples

wrapping

import std/ui/geometry
import std/ui/layout
import std/ui/scene
import std/ui/text

fn lines_of(s: str, l: text.Layout) {
    var at = 0
    for line in l.lines {
        let n = line.glyphs.len
        print("  [{s[at..at + n]}] width={line.width} baseline={line.y}")
        at += n
    }
}

/// Wraps a sentence at two widths, measures it, reuses the cached layout and paints it into a scene.
fn main() {
    var sys = text.default_system()
    let s = "The quick brown fox jumps over the lazy dog"

    for width in [f32(180.0), f32(90.0)] {
        let l = text.wrap(mut sys, s, 16.0, width)
        print("width {width}: {l.lines.len} lines, {l.width} by {l.height}")
        lines_of(s, l)
    }

    let one_line = text.measure(mut sys, s, 16.0, layout.unbounded)
    print("unbounded: {one_line.width} by {one_line.height}")
    print("layouts held: {text.cached(sys)}")
    text.end_frame(mut sys)
    print("layouts after end_frame: {text.cached(sys)}")

    let l = text.wrap(mut sys, "Hi\nthere", 16.0, layout.unbounded)
    var sc = scene.empty(200, 100, geometry.rgb(0x202020))
    text.paint(mut sys, mut sc, l, geometry.point(8.0, 8.0), geometry.rgb(0xffffff), 1.0)
    print("sprites painted: {sc.sprites.len}")
    print("atlas has pending rows: {scene.dirty(sys.atlas)}")
}

paint

pub fn paint(mut sys: System, mut sc: scene.Scene, l: Layout, origin: geometry.Point, color: geometry.Color, scale: f32)

Pushes one sprite onto sc for each visible glyph of l, tinted color, with the layout's top left at origin and everything multiplied by scale.

Glyphs are rasterised at l.size * scale pixels, rounded to 1/64 pixel, the first time they are drawn, and packed into the atlas with a one-pixel empty border. Each glyph's pen position is rounded to a whole device pixel. A glyph that does not fit in the atlas is left out, and so is every glyph packed after it until end_frame.

wrap

pub fn wrap(mut sys: System, s: str, size: f32, width: f32) -> Layout

Returns s laid out at size pixels in lines no wider than width pixels, or in lines that end only at \n when width is infinite.

Each code point is drawn by the first font that has it, else by the first font's glyph 0, and advances the pen by the glyph's advance plus the kern adjustment to the glyph before it when both come from one font. A line breaks after a space, so spaces at the end of a line hang past its width, and a word wider than width breaks between glyphs. \n ends a line, a \r is ignored, and a layout always has at least one line.

The result is kept until end_frame, so laying out the same s, size and width again returns it.
