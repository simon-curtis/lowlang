std/time/calendar

#time

import std/time/calendar

The proleptic Gregorian calendar the date types share: leap_year, days_in_month, days_from_civil and civil_from_days from the Unix epoch, weekday_index, and the English month and weekday names, full and short.

Imports std/math.

Examples

month_grid

import std/text
import std/time/calendar

fn print_month(year: i64, month: i64) {
    let title = "{calendar.month_names()[month - 1]} {year}"
    let pad = (20 - title.len) / 2
    print("{text.repeat(" ", pad)}{title}")

    var heading = ""
    for name in calendar.short_weekday_names() {
        heading = "{heading}{name[0..2]} "
    }
    print(text.trim_end(heading))

    let first = calendar.days_from_civil(year, month, 1)
    let offset = calendar.weekday_index(first)
    let length = calendar.days_in_month(year, month)
    var row = text.repeat("   ", offset)
    var column = offset
    for day in 1..length + 1 {
        row = "{row}{text.pad_left("{day}", 2, ' ')} "
        column += 1
        if column == 7 {
            print(text.trim_end(row))
            row = ""
            column = 0
        }
    }
    if column > 0 {
        print(text.trim_end(row))
    }
}

/// Prints calendar pages from the day counts of the proleptic Gregorian calendar.
fn main() {
    print_month(2026, 11)
    print("")
    print_month(2028, 2)

    for year in [1900, 2000, 2026, 2028, 2100] {
        print("{year} leap {calendar.leap_year(year)}, february has {calendar.days_in_month(year, 2)} days")
    }

    let days = calendar.days_from_civil(2026, 11, 13)
    let civil = calendar.civil_from_days(days + 100)
    print("2026-11-13 is day {days} since 1970-01-01")
    print("100 days later is {civil.year}-{civil.month}-{civil.day}")
    print("weekday names: {calendar.weekday_names()}")
}

Civil

pub struct Civil {The year, which can be zero or negative.    pub year: i64The month, from 1 for January to 12 for December.    pub month: i64The day of the month, from 1.    pub day: i64
}

A year, month and day of the proleptic Gregorian calendar.

Functions

civil_from_days

pub fn civil_from_days(days: i64) -> Civil

Returns the date that is days days after 1970-01-01, or before it when days is negative.

days_from_civil

pub fn days_from_civil(year: i64, month: i64, day: i64) -> i64

Returns the number of days from 1970-01-01 to the given date, negative before it.

The month and day are not range checked.

days_in_month

pub fn days_in_month(year: i64, month: i64) -> i64

Returns the number of days in month month, from 1 for January, of year.

leap_year

pub fn leap_year(year: i64) -> bool

Returns whether year is a leap year in the proleptic Gregorian calendar.

pub fn weekday_index(days: i64) -> i64
