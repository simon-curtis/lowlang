import std/time/date
import std/time/datetime
import std/time/duration
import std/time/fault
import std/time/timeofday
import std/time/timestamp
import std/time/tzdata
import std/time/zone
import std/time/zoned

fn advice(f: fault.Fault) -> str {
    match f {
        OutOfRange(what) => "ask again: {what} has no such value"
        Invalid(what) => "ask again: {what} does not exist"
        Syntax(at) => "highlight byte {at}"
        Overflow => "cap the value"
        UnknownZone(name) => "offer a zone list instead of {name}"
        Malformed(what) => "report corrupt data: {what}"
        Unsupported(what) => "tell the user {what} cannot be used"
        Skipped => "ask for another time, that local time never happens"
        Ambiguous => "ask which of the two occurrences"
        OffsetConflict => "ask which of offset and zone to trust"
    }
}

fn london(name: str) -> zone.Zone ! fault.Fault {
    zone.load_from(name, tzdata.source())?
}

/// Triggers each kind of fault with ordinary input mistakes and shows the message and the reaction a caller would choose.
fn main() {
    var found: []fault.Fault = []
    let epoch_date = date.epoch()

    let day = date.of(2026, 2, 30) catch {
        f => {
            push(mut found, f)
            epoch_date
        }
    }
    let clock = timeofday.hm(24, 0) catch {
        f => {
            push(mut found, f)
            timeofday.midnight()
        }
    }
    let typed = datetime.parse("2026-11-13 9:00") catch {
        f => {
            push(mut found, f)
            datetime.epoch()
        }
    }
    let late = timestamp.plus(timestamp.latest(), duration.seconds(1)) catch {
        f => {
            push(mut found, f)
            timestamp.epoch()
        }
    }
    let nowhere = london("Mars/Base") catch {
        f => {
            push(mut found, f)
            zone.utc()
        }
    }
    let garbled = zone.decode("Broken", bytes("not a tzif file")) catch {
        f => {
            push(mut found, f)
            zone.utc()
        }
    }
    let wall = datetime.format(datetime.epoch(), "%c") catch {
        f => {
            push(mut found, f)
            ""
        }
    }
    let tz = london("Europe/London") or zone.utc()
    let spring = datetime.of(2026, 3, 29, 1, 30, 0, 0) or datetime.epoch()
    let autumn = datetime.of(2026, 10, 25, 1, 30, 0, 0) or datetime.epoch()
    let gap = zoned.strict(spring, tz) catch {
        f => {
            push(mut found, f)
            zoned.of(timestamp.epoch(), tz) or return
        }
    }
    let twice = zoned.strict(autumn, tz) catch {
        f => {
            push(mut found, f)
            zoned.of(timestamp.epoch(), tz) or return
        }
    }
    let conflict = zoned.parse_using("2026-07-01T12:00:00+05:00[Europe/London]", london) catch {
        f => {
            push(mut found, f)
            zoned.of(timestamp.epoch(), tz) or return
        }
    }

    for f in found {
        print("{fault.to_string(f)}: {advice(f)}")
    }
}
