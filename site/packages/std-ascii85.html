Imports std/text.

Examples

frames

import std/ascii85
import std/text

fn describe(e: ascii85.Error) -> str {
    match e {
        BadCharacter(at) => "bad character at {at}"
        BadLength => "bad length"
        BadFrame => "bad frame"
        Overflow(at) => "overflow at {at}"
    }
}

fn show(s: str) {
    let data = ascii85.decode(s) catch {
        e => {
            print("{s}: {describe(e)}")
            return
        }
    }
    print("{s}: {text.encode_hex(data)}")
}

/// Wraps bytes in an Adobe frame for a PostScript stream, reads framed text back, and writes a binary key as Z85.
fn main() {
    let body = bytes("Man is distinguished")
    let framed = ascii85.encode_adobe(body)
    print(framed)
    let back = ascii85.decode(framed) or bytes("")
    print(str(back))

    let zeros = text.decode_hex("00000000000000000102") or bytes("")
    print(ascii85.encode(zeros))

    show("<~87cURDZ~>")
    show("  <~ 87cUR DZ ~>  ")
    show("<~87cURDZ")
    show("<~87c~RDZ~>")
    show("<~s8W-#~>")
    show("<~8~>")

    let key = text.decode_hex("864fd26fb559f75b") or bytes("")
    let z85 = ascii85.encode_z85(key) or ""
    print(z85)
    let raw = ascii85.decode_z85(z85) or bytes("")
    print(text.encode_hex(raw))
    let odd = ascii85.encode_z85(bytes("abc")) catch {
        e => {
            print("z85 of 3 bytes: {describe(e)}")
            return
        }
    }
    print(odd)
}

Functions

decode

pub fn decode(s: str) -> []u8 ! Error

Decodes Ascii85 text, with or without the Adobe <~ and ~> delimiters. Whitespace is ignored, including around the delimiters. z stands for four zero bytes and is valid only between groups. A final partial group of 2 to 4 characters yields one byte fewer than its length.

Fails with BadFrame when only one delimiter is present, with BadCharacter at the index in s of a byte outside ! to u, with BadLength when the final group has one character, and with Overflow when a group exceeds 0xFFFFFFFF.

decode_z85

pub fn decode_z85(s: str) -> []u8 ! Error

Decodes Z85 text. The length must be a multiple of 5 and whitespace is not skipped. Fails with BadLength otherwise, with BadCharacter at the index of the first byte outside the alphabet, and with Overflow when a group exceeds 0xFFFFFFFF.

encode

pub fn encode(data: []u8) -> str

Returns data encoded as Ascii85 without delimiters. Each group of four bytes becomes five characters in ! to u, and a group of four zero bytes becomes z. A final group of n bytes gives n + 1 characters.

encode_adobe

pub fn encode_adobe(data: []u8) -> str

Returns data encoded as Ascii85 inside the Adobe <~ and ~> delimiters.

encode_z85

pub fn encode_z85(data: []u8) -> str ! Error

Returns data encoded as Z85 (ZeroMQ), which uses the 85-character alphabet 0-9a-zA-Z.-:+=^!/*?&<>()[]{}@%$#. Fails with BadLength when data.len is not a multiple of 4.
