Point, Size, Rect, Edges, Corners and Color, all in f32.

Examples

damage

import std/ui/geometry

fn show(label: str, r: geometry.Rect) {
    print("{label}: x={r.x} y={r.y} w={r.width} h={r.height} empty={geometry.is_empty(r)}")
}

/// Tracks the region a frame redraws, hit tests a button with its padding and converts logical pixels to device pixels.
fn main() {
    let window = geometry.rect(0.0, 0.0, 200.0, 120.0)
    let button = geometry.rect(20.0, 40.0, 80.0, 30.0)
    let tooltip = geometry.rect(90.0, 60.0, 150.0, 40.0)

    show("tooltip clipped", geometry.intersect(tooltip, window))
    show("damage", geometry.union(button, tooltip))
    show("disjoint", geometry.intersect(button, geometry.rect(150.0, 0.0, 10.0, 10.0)))

    let hit_area = geometry.outset(button, geometry.edges(4.0))
    show("hit area", hit_area)
    for p in [geometry.point(15.9, 38.0), geometry.point(16.0, 38.0), geometry.point(104.0, 74.0), geometry.point(103.9, 73.9)] {
        print("({p.x}, {p.y}) hits={geometry.contains(hit_area, p)}")
    }

    let padding = geometry.Edges { top: 4.0, right: 12.0, bottom: 4.0, left: 12.0 }
    let label = geometry.inset(button, padding)
    show("label box", label)
    print("padding: {geometry.horizontal(padding)} across, {geometry.vertical(padding)} down")

    let device = geometry.scale(button, 2.0)
    show("device", device)
    let radii = geometry.scale_corners(geometry.corners(6.0), 2.0)
    print("device radius: {radii.top_left}")

    let accent = geometry.rgb(0xff8000)
    let faded = geometry.with_alpha(accent, 0.5)
    print("accent: r={accent.r} g={accent.g} b={accent.b} a={accent.a}")
    print("faded alpha: {faded.a}")
    print("right edge {geometry.right(button)}, bottom edge {geometry.bottom(button)}")
}
