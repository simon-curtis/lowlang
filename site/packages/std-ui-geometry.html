#ui

import std/ui/geometry

Point, Size, Rect, Edges, Corners, Color and Path, all in f32, and the flattening and stroking of paths into polygons.

Path

pub struct Path {
    ...
}

A shape of contours made of lines and Bézier curves, in logical pixels.

affine

pub fn affine(p: Path, a: f32, b: f32, c: f32, d: f32, e: f32, f: f32) -> Path

Returns p with every point (x, y) moved to (a * x + c * y + e, b * x + d * y + f).

close

pub fn close(mut p: Path)

Closes the current contour with a line back to its start.

cubic_to

pub fn cubic_to(mut p: Path, first: Point, second: Point, to: Point)

Adds a cubic Bézier curve from the current point to to with the control points first and second.

flatten

pub fn flatten(p: Path, scale: f32) -> [][]Point

Returns the contours of p multiplied by scale as polygons, each curve split into lines that stay within a twentieth of a pixel of it. Every polygon is closed implicitly.

line_to

pub fn line_to(mut p: Path, to: Point)

Adds a straight line from the current point to to.

move_to

pub fn move_to(mut p: Path, to: Point)

Starts a new contour at to.

quadratic_to

pub fn quadratic_to(mut p: Path, control: Point, to: Point)

Adds a quadratic Bézier curve from the current point to to pulled toward control.

segments

pub fn segments(p: Path) -> []Segment

Returns the steps of p in the order they were added.

stroke

pub fn stroke(p: Path, width: f32, scale: f32) -> [][]Point

Returns polygons whose union by the nonzero rule covers the contours of p, multiplied by scale, traced by a pen width logical pixels wide with round joins and caps. Only closed contours join their last point to their first.

Point

extern pub struct Point {The distance to the right.    pub x: f32The distance down.    pub y: f32
}

A position in pixels from the top left, x to the right and y down.

wave

pub fn wave(origin: Point, width: f32, thickness: f32) -> Path

Returns an open path of quadratic curves running width to the right from origin, swinging thickness either side of the line thickness / 2 below it, one whole wave every six thicknesses: the centre line of a wavy underline thickness thick whose top is at origin.

inset

Returns r with each side moved inward by its edge in e, never smaller than empty.

intersect

Returns the overlap of a and b, a rectangle of zero width or height when they do not overlap.

is_empty

pub fn is_empty(r: Rect) -> bool

Returns true when r covers no area.

offset

outset

pub fn outset(r: Rect, e: Edges) -> Rect

right

pub fn right(r: Rect) -> f32

Returns the x coordinate of the right edge of r.

scale

pub fn scale(r: Rect, factor: f32) -> Rect

Returns r with every coordinate multiplied by factor, as when logical pixels become device pixels.

union

pub fn union(a: Rect, b: Rect) -> Rect

Returns the smallest rectangle that holds both a and b.
