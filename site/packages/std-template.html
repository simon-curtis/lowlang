std/template

#web #text

import std/template

A text template engine over a Data model (to_data[T] through serde): {{ expr }}, if, for, include, block and extends, let, trim markers, filters (built in and registered), HTML autoescaping, depth and output limits, and errors with line and column. A Set is read-only while rendering, so green threads share it.

Imports std/base64, std/html, std/io, std/json, std/math, std/serde, std/text, std/time/fault, std/time/timestamp, std/time/zone, std/time/zoned, std/uri.

Examples

filters

import std/template
import std/text

fn show(mut views: template.Set, source: str, data: template.Data) {
    views.add("inline", source) catch {
        e => {
            print("error {template.describe(e)}")
            return
        }
    }
    let out = template.render(views, "inline", data) catch {
        e => {
            print("error {template.describe(e)}")
            return
        }
    }
    print(out)
}

/// Renders plain text from JSON data with built-in and registered filters, and reports template errors with their position.
fn main() {
    var views = template.set(template.Options { escape: None })
    views.add_filter("initials", fn(subject, args) -> template.Data ! template.FilterError {
        match subject {
            Str(name) => {
                var letters: []str = []
                for word in text.fields(name) {
                    push(mut letters, text.upper(word[..1]))
                }
                template.Data.Str(text.join(letters, ""))
            }
            _ => fail template.FilterError.Invalid("expects text")
        }
    })
    views.add("card", "{{{{ name | initials }}}}: {{{{ name | upper }}}}, {{{{ tags | join(\", \") }}}} ({{{{ tags | len }}}} {{{{ \"tag\" }}}}{{{{ tags | pluralize }}}})\n{{{{ nickname | default(\"no nickname\") }}}}; {{{{ profile.city | default(\"unknown city\") }}}}") or return

    let data = template.from_json("{{\"name\": \"ada king lovelace\", \"tags\": [\"math\", \"engines\"], \"nickname\": null, \"profile\": {{\"zip\": 12}}}}") catch {
        e => {
            print("bad data")
            return
        }
    }
    print(template.render(views, "card", data) or "failed")
    show(mut views, "{{{{ profile | json }}}} {{{{ \"a b&c\" | url }}}}", data)
    show(mut views, "{{{{ for tag in tags }}}}{{{{ @index }}}}:{{{{ tag }}}}{{{{ if !@last }}}},{{{{ end }}}}{{{{ end }}}}", data)

    show(mut views, "{{{{ name | initials }}}}", template.Data.Map(["name": template.Data.Int(5)]))
    show(mut views, "line one\n{{{{ if tags }}}}never closed", data)

    var strict = template.set(template.Options { strict: true })
    strict.add("t", "{{{{ profile.city }}}}") or return
    let strict_out = template.render(strict, "t", data) catch {
        e => {
            print("strict {template.describe(e)}")
            return
        }
    }
    print(strict_out)
}

invoice

import std/template

struct Line {
    item: str
    quantity: i64
    price: f64
    note: str?
}

struct Invoice {
    number: i64
    customer: str
    lines: []Line
    paid: bool
}

/// Renders an HTML invoice from a struct through a layout template that child templates extend, with autoescaping.
fn main() {
    var views = template.set(template.Options {})
    views.add("layout.html", "<html>\n<title>{{{{ block \"title\" }}}}Untitled{{{{ end }}}}</title>\n<body>\n{{{{ block \"body\" }}}}{{{{ end }}}}\n{{{{ include \"footer.html\" }}}}\n</body>\n</html>") catch {
        e => {
            print(template.describe(e))
            return
        }
    }
    views.add("footer.html", "<footer>Thank you, {{{{ customer }}}}.</footer>") or return
    views.add("invoice.html", "{{{{ extends \"layout.html\" }}}}\n{{{{ block \"title\" }}}}Invoice #{{{{ number }}}}{{{{ end }}}}\n{{{{ block \"body\" }}}}<h1>{{{{ customer }}}}</h1>\n<ul>\n{{{{- for line in lines }}}}\n  <li>{{{{ @index + 1 }}}}/{{{{ @length }}}} {{{{ line.item }}}} x{{{{ line.quantity }}}}{{{{ if line.note }}}} ({{{{ line.note | lower }}}}){{{{ end }}}}{{{{ if @last }}}} [last]{{{{ end }}}}</li>\n{{{{- else }}}}\n  <li>no lines</li>\n{{{{- end }}}}\n</ul>\n{{{{ if paid }}}}PAID{{{{ else }}}}DUE{{{{ end }}}}\n{{{{ end }}}}") or return

    let invoice = Invoice {
        number: 1042,
        customer: "Ada <Lovelace> & Co",
        lines: [
            Line { item: "Notebook", quantity: 3, price: 4.5, note: none },
            Line { item: "Pen \"fine\"", quantity: 10, price: 1.25, note: "BULK" },
        ],
        paid: false,
    }
    let page = template.render(views, "invoice.html", template.to_data(invoice)) catch {
        e => {
            print(template.describe(e))
            return
        }
    }
    print(page)

    let empty = Invoice { number: 1043, customer: "Bob", lines: [], paid: true }
    print(template.render(views, "invoice.html", template.to_data(empty)) or "failed")

    let missing = template.render(views, "receipt.html", template.Data.Null) catch {
        e => {
            print(template.describe(e))
            return
        }
    }
    print(missing)
}

Settings of a Set.

set

pub fn set(options: Options) -> Set

Returns an empty Set with options.

Set

pub struct Set {
    ...
}

A collection of named templates and filters that render together. A Set is only read while rendering, so green threads may share one once it is built. include and extends look templates up by name in the same Set.

add

pub fn add(mut s: Set, name: str, source: str) ! Error

Compiles source and adds it to s under name, replacing any template of that name. Fails with Error.Parse when the source is not a valid template.

add_filter

pub fn add_filter(mut s: Set, name: str, f: fn(Data, []Data) -> Data ! FilterError)

Registers the filter function f under name for use as value | name(arguments). f receives the subject and the argument list. It takes precedence over a built-in filter of the same name, and a FilterError it returns becomes an Error.Render.

has

pub fn has(s: Set, name: str) -> bool

Returns whether s has a template named name.

render

pub fn render(s: Set, name: str, data: Data) -> str ! Error

Renders the template name of s with data and returns the output. data is the root scope: names in the template are looked up in it when it is a Data.Map. Fails with NotFound when name is not in s, and with Render, TooDeep or TooLarge while rendering.

render_to

pub fn render_to(s: Set, name: str, data: Data, out: io.Stream) ! Error

Renders the template name of s with data and writes the output to out. Output is written in chunks as it is produced, so part of it may already be written when an error is returned. Fails as render does and with Write when out fails.

Functions

compile

pub fn compile(name: str, source: str) -> Template ! Error

Compiles source into a Template named name without adding it to a Set. name is used in the errors. Fails with Error.Parse for a syntax error such as an unclosed tag, an unbalanced if, for or block, a repeated extends or block name, or nesting deeper than 100 levels.

from_json

pub fn from_json(source: str) -> Data ! serde.Error

Parses the JSON text source into Data. Objects become Map, arrays List and numbers Int or Float. Fails with a serde.Error when source is not valid JSON.

list

pub fn list(items: []Data) -> Data

Returns the Data list of items.

map

pub fn map(entries: [str: Data]) -> Data

Returns the Data map of entries.

render_source

pub fn render_source(source: str, data: Data) -> str ! Error

Compiles source and renders it once with data, with default Options.

to_data

pub fn to_data[T](v: T) -> Data

Converts any serialisable value v into Data. Structs and maps become Map, sequences become List, an enum variant without a payload becomes the Str of its name and one with a payload becomes a Map with the variant name as its only key. Byte strings become base64 Str and an unsigned integer above the i64 range becomes a Float.
