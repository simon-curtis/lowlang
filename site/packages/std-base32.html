std/base32

#encoding

import std/base32

RFC 4648 base32 with the standard and extended-hex alphabets: encode, encode_unpadded, decode (case-insensitive, padding optional, failing at the first bad character or on an impossible length), encode_hex and decode_hex.

Imports std/text.

secret

import std/base32
import std/text

fn grouped(s: str) -> str {
    var parts: []str = []
    var at = 0
    for at < s.len {
        let end = if at + 4 < s.len { at + 4 } else { s.len }
        push(mut parts, s[at..end])
        at = end
    }
    text.join(parts, " ")
}

fn check(entered: str) {
    let cleaned = text.replace(entered, " ", "")
    let key = base32.decode(cleaned) catch {
        BadCharacter(at) => {
            print("{entered}: bad character at {at}")
            return
        }
        BadLength => {
            print("{entered}: bad length")
            return
        }
    }
    print("{entered}: {text.encode_hex(key)}")
}

/// Shows an authenticator secret in groups, reads back what a user types, and compares the padded, unpadded and hex alphabets.
fn main() {
    let key = text.decode_hex("48656c6c6f21deadbeef") or bytes("")
    let secret = base32.encode_unpadded(key)
    print(grouped(secret))

    check(grouped(secret))
    check(text.lower(grouped(secret)))
    check("JBSWY3D1")
    check("JBSWY3DPE")

    let word = bytes("foobar")
    print(base32.encode(word))
    print(base32.encode_unpadded(word))
    print(base32.encode_hex(word))
    let back = base32.decode_hex(base32.encode_hex(word)) or bytes("")
    print(str(back))
}

Functions

decode

pub fn decode(s: str) -> []u8 ! Error

Decodes RFC 4648 Base32 text, with or without = padding. Lowercase letters are accepted. Any number of trailing = is stripped before decoding, so an = anywhere else is a BadCharacter.

Fails with BadLength when the unpadded length leaves a remainder of 1, 3 or 6 characters modulo 8, and with BadCharacter at the index of the first byte outside the alphabet. Bits left over in the last character are ignored.

decode_hex

pub fn decode_hex(s: str) -> []u8 ! Error

Decodes Base32 text written with the extended hex alphabet 0-9A-V, with or without = padding. The failure cases, trailing padding and case handling are those of decode.

encode

pub fn encode(data: []u8) -> str

Returns data encoded as Base32 with the RFC 4648 alphabet A-Z2-7, padded with = to a multiple of 8 characters.

encode_hex

pub fn encode_hex(data: []u8) -> str

Returns data encoded as Base32 with the extended hex alphabet 0-9A-V, padded with = to a multiple of 8 characters.

encode_unpadded

pub fn encode_unpadded(data: []u8) -> str

Returns data encoded as Base32 with the RFC 4648 alphabet A-Z2-7 and no = padding.
