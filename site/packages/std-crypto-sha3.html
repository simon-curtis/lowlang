Examples

digests

import std/crypto/hash
import std/crypto/sha3
import std/text

/// Hashes with SHA3-256, authenticates with HMAC-SHA3-256, and reads an extendable SHAKE256 stream in two pieces.
fn main() {
    let sha3_256 = sha3.sha3_256()
    print("sha3-256 {text.encode_hex(hash.sum(sha3_256, bytes("abc")))}")
    print("sha3-512 {text.encode_hex(hash.sum(sha3.sha3_512(), bytes("")))}")

    let tag = hash.hmac(sha3_256, bytes("key"), bytes("message"))
    print("hmac {text.encode_hex(tag)} ({tag.len} bytes)")

    let stream = sha3.shake256()
    sha3.absorb(stream, bytes("a"))
    sha3.absorb(stream, bytes("bc"))
    let first = text.encode_hex(sha3.squeeze(stream, 8))
    let second = text.encode_hex(sha3.squeeze(stream, 8))
    print("shake256 {first} {second}")

    let whole = sha3.shake256()
    sha3.absorb(whole, bytes("abc"))
    let joined = text.encode_hex(sha3.squeeze(whole, 16))
    print("one squeeze matches: {joined == "{first}{second}"}")
}

Types

Shake

pub struct Shake {
    ...
}

A SHAKE extendable-output sponge (FIPS 202).

Input is absorbed with absorb; the first squeeze ends absorption and returns output of any length.

absorb

pub fn absorb(s: *Shake, data: []u8)

Feeds data into the sponge. A call after the first squeeze has no effect.

squeeze

pub fn squeeze(s: *Shake, n: i64) -> []u8

Returns the next n bytes of output.

The first call ends absorption. Successive calls continue the same output stream. An n of 0 or less returns an empty slice.
