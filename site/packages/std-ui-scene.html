Quads, shadows and sprites with their clip rectangles and draw order, and batching.

Imports std/ui/geometry.

Examples

card

import std/ui/geometry
import std/ui/scene

fn describe(kind: scene.Kind) -> str {
    match kind {
        Quads => "quads"
        Shadows => "shadows"
        Sprites => "sprites"
    }
}

fn show(s: scene.Scene) {
    for b in scene.batches(s) {
        print("  {describe(b.kind)} from {b.first}, count {b.count}")
    }
}

/// Draws a card with a shadow, a clipped body and glyph sprites, then lists the batches the renderer draws and the atlas rows it uploads.
fn main() {
    var s = scene.empty(400, 300, geometry.rgb(0xf0f0f0))
    let card = geometry.rect(40.0, 40.0, 200.0, 120.0)

    scene.push_shadow(mut s, scene.Shadow { bounds: card, color: geometry.rgba(0x00000060), corner_radii: geometry.corners(8.0), sigma: 6.0 })
    scene.push_quad(mut s, scene.Quad {
        bounds: card
        background: geometry.rgb(0xffffff)
        border_color: geometry.rgb(0xcccccc)
        corner_radii: geometry.corners(8.0)
        border_widths: geometry.edges(1.0)
    })

    scene.push_clip(mut s, geometry.inset(card, geometry.edges(10.0)))
    let inner = scene.clip(s)
    print("clip inside card: x={inner.x} y={inner.y} w={inner.width} h={inner.height}")
    scene.push_clip(mut s, geometry.rect(0.0, 0.0, 100.0, 100.0))
    let narrow = scene.clip(s)
    print("clip after second push: w={narrow.width} h={narrow.height}")

    var atlas = scene.atlas(64)
    scene.clean(mut atlas)
    print("atlas dirty after clean: {scene.dirty(atlas)}")
    scene.write(mut atlas, 4, 10, 3, 2, [255, 128, 0, 0, 128, 255])
    print("atlas rows to upload: {atlas.dirty_top} to {atlas.dirty_bottom}")
    let tile = scene.tile(atlas, 4, 10, 3, 2)

    for i in 0..3 {
        scene.push_sprite(mut s, scene.Sprite { bounds: geometry.rect(52.0 + f32(i) * 8.0, 60.0, 3.0, 2.0), color: geometry.rgb(0x000000), tile })
    }
    scene.pop_clip(mut s)
    scene.pop_clip(mut s)
    scene.pop_clip(mut s)
    let restored = scene.clip(s)
    print("clip after pops: w={restored.width} h={restored.height}")

    scene.push_quad(mut s, scene.Quad { bounds: geometry.rect(60.0, 120.0, 80.0, 24.0), background: geometry.rgb(0x2060ff) })
    scene.push_quad(mut s, scene.Quad { bounds: geometry.rect(150.0, 120.0, 80.0, 24.0), background: geometry.rgb(0xff6020) })

    print("batches:")
    show(s)
    print("sprite clip height: {s.sprites[0].clip.height}")
}

clip after second push: w=50 h=50
atlas dirty after clean: false
atlas rows to upload: 10 to 12
clip after pops: w=400 h=300

  shadows from 0, count 1
  quads from 0, count 1
  sprites from 0, count 3
  quads from 1, count 2
sprite clip height: 50

Atlas

pub struct Atlas {The width and height in pixels.    pub size: i64One coverage byte per pixel, rows from the top.    pub pixels: []u8The first row changed since clean, or size when none has.    pub dirty_top: i64One past the last row changed since clean, or 0 when none has.    pub dirty_bottom: i64
}

A square single-channel image of glyph coverage that sprites sample, with the band of rows changed since the GPU last copied it.

clean

pub fn clean(mut a: Atlas)

Marks every row as copied.

clear

pub fn clear(mut a: Atlas)

Zeroes every pixel and marks the whole atlas changed.

dirty

pub fn dirty(a: Atlas) -> bool

Returns true when rows have changed since the last clean.

tile

pub fn tile(a: Atlas, x: i64, y: i64, width: i64, height: i64) -> geometry.Rect

Returns the fractions of the atlas that a width by height pixel tile at x, y covers, for Sprite.tile.

write

pub fn write(mut a: Atlas, x: i64, y: i64, width: i64, height: i64, coverage: []u8)

Copies coverage, width by height bytes in rows, into the atlas with its top-left pixel at x, y, and marks those rows changed.

The primitives of one frame in device pixels, in the order they were pushed.

batches

pub fn batches(s: Scene) -> []Batch

Returns the runs to draw, in order, so that primitives of every kind draw in the order they were pushed.

Consecutive primitives of one kind share a run, so a scene of quads under text is two batches however many of each it holds.

clip

pub fn clip(s: Scene) -> geometry.Rect

Returns the rectangle primitives are currently clipped to: the intersection of every clip pushed and not popped, and the target.

pop_clip

pub fn pop_clip(mut s: Scene)

Restores the clip in force before the last push_clip. The target's own clip is never removed.

push_clip

pub fn push_clip(mut s: Scene, r: geometry.Rect)

Narrows the clip to its intersection with r until the matching pop_clip.

push_quad

pub fn push_quad(mut s: Scene, q: Quad)

Adds q above everything pushed so far, clipped to the current clip.

push_shadow

pub fn push_shadow(mut s: Scene, sh: Shadow)

Adds sh above everything pushed so far, clipped to the current clip.

push_sprite

pub fn push_sprite(mut s: Scene, sp: Sprite)

Adds sp above everything pushed so far, clipped to the current clip.
