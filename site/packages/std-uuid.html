std/uuid

#data

import std/uuid

Random v4 and time-ordered v7 UUIDs, format, parse (reports the offset of the first bad character), version, timestamp for v7, equal and nil.

Imports std/crypto, std/text, std/time.

Examples

identifiers

import std/uuid

fn describe(raw: str) {
    let id = uuid.parse(raw) catch {
        Length => {
            print("{raw}: expected 36 characters, got {raw.len}")
            return
        }
        BadDigit(at) => {
            print("{raw}: bad character at {at}")
            return
        }
    }
    let when = uuid.timestamp(id)
    match when {
        some(ms) => print("{uuid.format(id)}: version {uuid.version(id)}, created at {ms} ms")
        none => print("{uuid.format(id)}: version {uuid.version(id)}, no timestamp")
    }
}

/// Parses identifiers from text, reads the creation time of a version 7 UUID, and checks the shape of freshly generated ones.
fn main() {
    describe("017F22E2-79B0-7CC3-98C4-DC0C0C07398F")
    describe("919108f7-52d1-4320-9bac-f847db4148a8")
    describe("00000000-0000-0000-0000-000000000000")
    describe("919108f7-52d1-4320-9bac-f847db4148")
    describe("919108f7_52d1-4320-9bac-f847db4148a8")
    describe("919108f7-52d1-4320-9bac-f847db4148ag")

    let upper = uuid.parse("017F22E2-79B0-7CC3-98C4-DC0C0C07398F") or uuid.nil()
    let lower = uuid.parse("017f22e2-79b0-7cc3-98c4-dc0c0c07398f") or uuid.nil()
    print("same identifier: {uuid.equal(upper, lower)}")
    print("not nil: {!uuid.equal(upper, uuid.nil())}")

    let random = uuid.v4()
    let text = uuid.format(random)
    let again = uuid.parse(text) or uuid.nil()
    print("v4 version {uuid.version(random)}, {text.len} characters, survives a round trip: {uuid.equal(random, again)}")
    print("v4 has no timestamp: {uuid.timestamp(random) == none}")

    let first = uuid.v7()
    let second = uuid.v7()
    let stamped = uuid.timestamp(first) != none
    print("v7 version {uuid.version(first)}, has a timestamp: {stamped}, distinct: {!uuid.equal(first, second)}")
}

A 128-bit universally unique identifier. bytes holds the 16 bytes in network order.

equal

pub fn equal(a: Uuid, b: Uuid) -> bool

Returns whether a and b hold the same 16 bytes.

format

pub fn format(u: Uuid) -> str

Returns u as 36 lowercase characters in the 8-4-4-4-12 hyphenated form.

timestamp

pub fn timestamp(u: Uuid) -> i64?

Returns the creation time of a version 7 UUID as Unix milliseconds, or none for any other version.

version

pub fn version(u: Uuid) -> i64

Returns the version number held in the high four bits of byte 6. The variant bits are not checked.

Functions

nil

pub fn nil() -> Uuid

Returns the nil UUID, whose 128 bits are all zero.

parse

pub fn parse(s: str) -> Uuid ! ParseError

Parses the 36-character hyphenated form, accepting upper- and lowercase digits. Fails with Length when s is not 36 bytes, and with BadDigit at the index of the first byte that is not a hexadecimal digit or not a hyphen at index 8, 13, 18 or 23. The version and variant are not checked.

v4

pub fn v4() -> Uuid

Returns a random version 4 UUID with the RFC 9562 variant bits. The 122 free bits come from std/crypto's random source.

v7

pub fn v7() -> Uuid

Returns a version 7 UUID that sorts by creation time. The first 48 bits are the Unix time in milliseconds read from the real-time clock, the version and variant bits follow RFC 9562, and the remaining bits are random. UUIDs made within one millisecond are not ordered among themselves.
