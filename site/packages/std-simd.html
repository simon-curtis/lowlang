Examples

scan_text

import std/bits
import std/simd

fn count_byte(data: []u8, c: u8) -> i64 {
    let needle: u8x32 = simd.splat(c)
    var total = 0
    var i = 0
    for i + 32 <= data.len {
        let chunk: u8x32 = simd.read(data, i)
        total += bits.popcount(simd.mask(chunk == needle))
        i += 32
    }
    for i < data.len {
        if data[i] == c {
            total += 1
        }
        i += 1
    }
    total
}

fn upper_ascii(data: []u8) {
    let lo: u8x32 = simd.splat('a')
    let hi: u8x32 = simd.splat('z')
    let shift: u8x32 = simd.splat(u8(32))
    var i = 0
    for i + 32 <= data.len {
        let chunk: u8x32 = simd.read(data, i)
        let is_lower = (chunk >= lo) & (chunk <= hi)
        let shifted = chunk - shift
        simd.write(data, i, simd.blend(is_lower, shifted, chunk))
        i += 32
    }
    for i < data.len {
        if data[i] >= 'a' && data[i] <= 'z' {
            data[i] -= 32
        }
        i += 1
    }
}

/// Counts lines and finds a field separator in a log buffer 32 bytes at a time, then upper-cases the ASCII letters of the buffer with a vector blend.
fn main() {
    let log = "boot: disk ready\nnet: link up\nnet: dhcp lease acquired\nauth: session opened for root\ncron: nightly job done\n"
    print("{log.len} bytes, {count_byte(bytes(log), '\n')} lines, {count_byte(bytes(log), 'n')} letter n")
    print("first colon at {simd.find_byte_str(log, 0, log.len, ':')}")
    print("tab at {simd.find_byte_str(log, 0, log.len, '\t')}")

    let section: u8x16 = simd.read_str(log, 0)
    let found = simd.mask(section == ':')
    let first: u8 = simd.lane(section, 0)
    print("colons in the first 16 bytes: mask {found}, first byte {first}")

    let data = bytes(log)
    upper_ascii(data)
    print(str(data[..29]))
}

Output

first colon at 4
tab at -1
colons in the first 16 bytes: mask 16, first byte 98
BOOT: DISK READY

any

pub fn any[V: Vector](v: V) -> bool

Returns true when the top bit of at least one lane of v is set.

Applied to the result of a lane-wise comparison, it tells whether any lane compared true.

blend

pub fn blend[V: Vector](choose: V, a: V, b: V) -> V

Returns a vector that takes each lane from a where the matching lane of choose is set and from b where it is clear.

choose is normally the result of a lane-wise comparison, whose lanes are all ones or all zeros. A lane of choose with only some bits set is not defined to select either input.

find_byte

pub fn find_byte(data: []u8, from: i64, to: i64, c: u8) -> i64

Returns the index of the first byte equal to c in data from index from up to but not including to, or -1 when there is none.

Compares 32 bytes at a time while at least 32 remain. Panics when the scan reaches an index past the end of data.

find_byte_str

pub fn find_byte_str(s: str, from: i64, to: i64, c: u8) -> i64

Returns the index of the first byte equal to c in s from index from up to but not including to, or -1 when there is none.

Compares 32 bytes at a time while at least 32 remain. Panics when the scan reaches an index past the end of s.

lane

pub fn lane[V: Vector, E](v: V, i: i64) -> E

Returns lane i of v, counting from 0.

Panics when i is negative or not less than the number of lanes.

mask

pub fn mask[V: Vector](v: V) -> u64

Returns a bit mask with one bit per lane of v, taken from the top bit of the lane.

Lane 0 is bit 0. Applied to the result of a lane-wise comparison, bit i is set when the comparison held in lane i.

max

pub fn max[V: Vector](a: V, b: V) -> V

Returns the lane-wise maximum of a and b.

min

pub fn min[V: Vector](a: V, b: V) -> V

Returns the lane-wise minimum of a and b.

read

pub fn read[V: Vector, E](data: []E, at: i64) -> V

Returns the vector of type V whose lanes are the elements of data starting at index at.

at counts elements of data, not bytes, and the read is bounds-checked, so it panics when the vector would extend past the end of data.

read_str

pub fn read_str[V: Vector](s: str, at: i64) -> V

Returns the vector of type V whose lanes are the bytes of s starting at byte index at.

The read is bounds-checked and panics when the vector would extend past the end of s.

shuffle

pub fn shuffle(table: u8x32, indices: u8x32) -> u8x32

Looks up the bytes of indices in table, 16 bytes at a time, and returns the 32 bytes found.

Within each 16-byte half, result byte i is the byte of the same half of table at position indices[i] & 15, or zero when the top bit of indices[i] is set. The lookup never crosses between the two halves.

shuffle16

pub fn shuffle16(table: u8x16, indices: u8x16) -> u8x16

Looks up the bytes of indices in table and returns the 16 bytes found.

Result byte i is table[indices[i] & 15], or zero when the top bit of indices[i] is set.

splat

pub fn splat[V: Vector, E](x: E) -> V

Returns a vector of type V with every lane set to x.

with_lane

pub fn with_lane[V: Vector, E](v: V, i: i64, x: E) -> V

Returns a copy of v with lane i set to x.

Panics when i is negative or not less than the number of lanes. v does not change.

write

pub fn write[V: Vector, E](data: []E, at: i64, v: V)

Stores the lanes of v into data starting at index at.

at counts elements of data, not bytes, and the write is bounds-checked, so it panics when the vector would extend past the end of data.
