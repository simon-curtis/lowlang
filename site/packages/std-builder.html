import std/builder

A Builder over a growing byte buffer, appended in place through mut: write, write_bytes, write_byte, write_char, write_i64, write_f64, write_line, write_repeat, write_join, len, capacity, reserve, truncate, reset, to_str and to_bytes.

Examples

invoice

import std/builder

struct Line {
    item: str
    quantity: i64
    cents: i64
}

fn money(mut b: builder.Builder, cents: i64) {
    b.write_i64(cents / 100)
    b.write_byte('.')
    if cents % 100 < 10 {
        b.write_byte('0')
    }
    b.write_i64(cents % 100)
}

/// Renders an invoice into one growing buffer, then reuses the buffer for a one-line summary.
fn main() {
    let lines = [
        Line { item: "keyboard", quantity: 1, cents: 8950 },
        Line { item: "cable", quantity: 3, cents: 499 },
        Line { item: "adapter", quantity: 2, cents: 1205 }
    ]
    var b = builder.with_capacity(64)
    b.write_line("INVOICE")
    b.write_repeat("-", 24)
    b.write_byte('\n')
    var total = 0
    for line in lines {
        let amount = line.quantity * line.cents
        total += amount
        b.write(line.item)
        b.write_repeat(" ", 12 - line.item.len)
        b.write_i64(line.quantity)
        b.write(" x ")
        money(mut b, amount)
        b.write_byte('\n')
    }
    b.write("total ")
    money(mut b, total)
    b.write_byte('\n')
    print(b.to_str())
    print("{b.len()} bytes written")

    b.reset()
    b.write_join(["keyboard", "cable", "adapter"], ", ")
    b.write(" -> ")
    b.write_char(0x20AC)
    money(mut b, total)
    print("{b.to_str()} ({b.len()} bytes, capacity kept: {b.capacity() >= 64})")

    b.truncate(8)
    print(b.to_str())
}

Types

Builder

pub struct Builder {
    ...
}

A growable byte buffer for assembling text and binary data without the cost of repeated string concatenation. A Builder is passed as mut. The buffer doubles when it fills. Obtain one from builder or with_capacity.

capacity

pub fn capacity(b: Builder) -> i64

Returns the number of bytes allocated, which is at least len.

len

pub fn len(b: Builder) -> i64

Returns the number of bytes written.

reserve

pub fn reserve(mut b: Builder, n: i64)

Ensures b can take n more bytes without growing, leaving its length unchanged.

reset

pub fn reset(mut b: Builder)

Empties b and keeps its allocation.

to_bytes

pub fn to_bytes(b: Builder) -> []u8

Returns the written bytes as a []u8 that shares storage with b. The slice is not a copy: writes made after reset or truncate overwrite the bytes it holds.

to_str

pub fn to_str(b: Builder) -> str

Returns a copy of the written bytes as a str.

truncate

pub fn truncate(mut b: Builder, n: i64)

Shortens b to n bytes and keeps its allocation. Nothing happens when n is negative or not below the current length.

write

pub fn write(mut b: Builder, s: str)

Appends the bytes of s.

write_byte

pub fn write_byte(mut b: Builder, c: u8)

Appends the single byte c.

write_bytes

pub fn write_bytes(mut b: Builder, data: []u8)

Appends the bytes of data.

write_char

pub fn write_char(mut b: Builder, c: u32)

Appends the UTF-8 encoding of the code point c, one to four bytes. The value is not validated: surrogates are encoded as they stand and a value above 0x1FFFFF gives a malformed byte sequence.

write_f64

pub fn write_f64(mut b: Builder, x: f64)

Appends x in the text form string interpolation gives an f64.

write_i64

pub fn write_i64(mut b: Builder, n: i64)

Appends n in decimal, with a leading - when negative.

write_join

pub fn write_join(mut b: Builder, parts: []str, sep: str)

Appends the strings in parts with sep between consecutive ones.

write_line

pub fn write_line(mut b: Builder, s: str)

Appends s followed by a newline byte.

write_repeat

pub fn write_repeat(mut b: Builder, s: str, n: i64)

Appends s n times. Nothing is appended when n is 0 or negative.
