Examples

shortest_path

import std/container/heap
import std/maps

struct Edge {
    to: str
    cost: i64
}

struct Visit {
    node: str
    distance: i64
}

/// Finds the cheapest route cost from one station to every other with Dijkstra's algorithm, using a heap ordered by distance.
fn main() {
    let network: [str: []Edge] = [
        "depot": [Edge { to: "market", cost: 7 }, Edge { to: "harbour", cost: 2 }],
        "harbour": [Edge { to: "market", cost: 3 }, Edge { to: "mill", cost: 9 }],
        "market": [Edge { to: "mill", cost: 4 }, Edge { to: "school", cost: 8 }],
        "mill": [Edge { to: "school", cost: 1 }],
        "school": []
    ]

    var queue = heap.heap[Visit](fn(a, b) a.distance < b.distance)
    var best: [str: i64] = [:]
    queue.insert(Visit { node: "depot", distance: 0 })

    for {
        let visit = queue.pop() ?? break
        if maps.has(best, visit.node) {
            continue
        }
        best[visit.node] = visit.distance
        for edge in network[visit.node] ?? [] {
            if !maps.has(best, edge.to) {
                queue.insert(Visit { node: edge.to, distance: visit.distance + edge.cost })
            }
        }
    }

    for node in maps.keys(best) {
        print("depot -> {node}: {best[node] ?? -1}")
    }

    var busiest = heap.heap[i64](fn(a, b) a > b)
    for load in [12, 40, 7, 33, 21] {
        busiest.insert(load)
    }
    print("peek {busiest.peek() ?? 0}, size {busiest.size()}")
    var top: []i64 = []
    for i in 0..3 {
        push(mut top, busiest.pop() ?? 0)
    }
    print("three busiest {top}")
}

Output

depot -> depot: 0
depot -> harbour: 2
depot -> market: 5
depot -> mill: 9
depot -> school: 10
peek 40, size 5
three busiest [40, 33, 21]

Types

Heap

pub struct Heap[T] {
    ...
}

A binary heap that yields its smallest item first, as decided by a less function.

Create one with heap. Mutating functions take the heap as mut.

insert

pub fn insert[T](mut h: Heap[T], item: T)

Adds item to h.

Sifts the item up with O(log n) calls to less.

peek

pub fn peek[T](h: Heap[T]) -> T?

Returns the item that pop would return, or none when h is empty.

Takes constant time.

pop

pub fn pop[T](mut h: Heap[T]) -> T?

Removes and returns the item that less ranks first, or none when h is empty.

Items that compare equal leave in no defined order. Sifting down takes O(log n) calls to less.

size

pub fn size[T](h: Heap[T]) -> i64

Returns the number of items in h.

Functions

heap

pub fn heap[T](less: fn(T, T) -> bool) -> Heap[T]

Returns an empty heap ordered by less.

less(a, b) returns true when a must leave the heap before b. It must be a strict weak ordering. For a max-heap, pass a function that compares in the opposite direction.
