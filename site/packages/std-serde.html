std/serde

#formats

import std/serde

The Writer and Reader protocols, write[T], read[T] and schema[T] (which the compiler derives for every type), and Field[T, F]. Readers share parse_integer, parse_unsigned and syntax_at, which builds the Syntax error for a line and column; the helpers the derived code calls are private. Tree, Node and Member are the value shape that CBOR, MessagePack, TOML, YAML and template data share: to_tree builds one from any value, tree_reader reads a T from one, and key_text and unsigned_tree hold the key text and the u64 range rule. A format maps its own value type V to a tree with node_of and a Shape[V] function, and back with value_of and a Rebuild[V].

Imports std/base64, std/text.

Examples

flatten

import std/serde
import std/text

struct Frame {
    prefix: str
    sequence: bool
    index: i64
}

struct State {
    frames: []*Frame
    key: str
    lines: []str
}

struct Limits {
    cpu: f64
    memory: u64
}

enum Mode {
    Local
    Remote(host: str, port: u16)
    Pool(str, str)
}

struct Service {
    name: str
    replicas: i32
    debug: bool
    mode: Mode
    ports: []u16
    limits: Limits
    labels: [str: str]
    token: []u8
    owner: str?
}

fn join(prefix: str, leaf: str) -> str {
    if prefix == "" {
        return leaf
    }
    "{prefix}.{leaf}"
}

fn name_for(s: *State) -> str {
    if s.frames.len == 0 {
        return "value"
    }
    let top = s.frames[s.frames.len - 1]
    if top.sequence {
        let at = top.index
        top.index += 1
        return join(top.prefix, "{at}")
    }
    join(top.prefix, s.key)
}

fn emit(s: *State, value: str) {
    let name = name_for(s)
    push(mut s.lines, "{name} = {value}")
}

fn open(s: *State, sequence: bool) {
    let name = if s.frames.len == 0 { "" } else { name_for(s) }
    push(mut s.frames, new(Frame { prefix: name, sequence, index: 0 }))
}

fn close(s: *State) {
    truncate(mut s.frames, s.frames.len - 1)
}

fn dotted_writer(s: *State) -> serde.Writer {
    serde.Writer {
        null: fn() { emit(s, "null") }
        boolean: fn(v) { emit(s, if v { "true" } else { "false" }) }
        integer: fn(n) { emit(s, "{n}") }
        unsigned: fn(n) { emit(s, "{n}") }
        float: fn(x) { emit(s, "{x}") }
        text: fn(v) { emit(s, "\"{v}\"") }
        binary: fn(b) { emit(s, "0x{text.encode_hex(b)}") }
        begin_seq: fn() { open(s, true) }
        end_seq: fn() { close(s) }
        begin_map: fn() { open(s, false) }
        key: fn(k) { s.key = k }
        end_map: fn() { close(s) }
        begin_struct: fn() { open(s, false) }
        field: fn(name) { s.key = name }
        end_struct: fn() { close(s) }
        unit_variant: fn(name) { emit(s, "\"{name}\"") }
        begin_variant: fn(name) {
            open(s, false)
            s.key = name
        }
        end_variant: fn() { close(s) }
    }
}

fn flatten[T](v: T) -> []str {
    let state = new(State { frames: [], key: "", lines: [] })
    serde.write(v, dotted_writer(state))
    state.lines
}

/// Writes derived values as dotted `path = value` lines through a custom `serde.Writer`.
fn main() {
    let service = Service {
        name: "api",
        replicas: 3,
        debug: false,
        mode: Remote("db.internal", 5432),
        ports: [80, 443],
        limits: Limits { cpu: 0.5, memory: 536870912 },
        labels: ["tier": "web", "team": "core"],
        token: [0xDE, 0xAD],
        owner: none,
    }
    for line in flatten(service) {
        print(line)
    }

    let local = Service { name: "dev", replicas: 1, debug: true, mode: Local, ports: [], limits: Limits { cpu: 1.0, memory: 1 }, labels: [:], token: [], owner: "ada" }
    for line in flatten(local) {
        print(line)
    }

    let pool = Mode.Pool("eu", "us")
    for line in flatten(pool) {
        print(line)
    }
    for line in flatten([1, 2]) {
        print(line)
    }
}

schema

import std/serde
import std/text

struct Address {
    street: str
    postcode: str?
}

enum Role {
    Guest
    Member(since: i64)
    Staff(Address, str)
}

struct Account {
    name: str
    age: u8 = 18
    emails: []str
    address: Address
    role: Role
    meta: [str: f64]
    parent: *Account?
}

fn kind_name(s: serde.Schema) -> str {
    match s.kind {
        Bool => "bool"
        Int => "i{s.bits}"
        Uint => "u{s.bits}"
        Float => "f{s.bits}"
        Text => "str"
        Binary => "bytes"
        Seq => "[{kind_name(s.items[0])}]"
        Map => "[{kind_name(s.items[0])}: {kind_name(s.items[1])}]"
        Optional => "{kind_name(s.items[0])}?"
        Struct => s.name
        Enum => s.name
        Ref => "-> {s.name}"
    }
}

fn show_properties(indent: str, properties: []serde.Property) {
    for p in properties {
        let marker = if p.required { "" } else { " (optional)" }
        print("{indent}{p.name}: {kind_name(p.schema)}{marker}")
    }
}

/// Lists the fields and variants that `serde.schema` derives for a struct, so a tool can document or validate a type without hand-written metadata.
fn main() {
    let account = serde.schema[Account]()
    print("{account.name} has {account.fields.len} fields")
    show_properties("  ", account.fields)

    for p in account.fields {
        match p.schema.kind {
            Struct => {
                print("{p.schema.name}:")
                show_properties("  ", p.schema.fields)
            }
            Enum => {
                print("{p.schema.name}:")
                for v in p.schema.variants {
                    let shape = if v.fields.len == 0 { "unit" } else if v.labelled { "labelled" } else { "positional" }
                    print("  {v.name} ({shape})")
                    show_properties("    ", v.fields)
                }
            }
            _ => {}
        }
    }

    let index = Account.age.index
    print("{Account.age.name} is field {index} of {account.fields.len}")
}

key_text

pub fn key_text(key: Node, at: i64) -> str ! Error

Returns the text of the map key key, which is how a map key of any scalar type matches a field name or a map key of a given type.

A Text is itself, an integer is its decimal digits, a boolean is true or false, a float is its default text and Null is null. Fails with Range at position at for an Overflow key and with Mismatch for a sequence, a map or an opaque value.

tree_reader

pub fn tree_reader(root: Node, base64_text: bool) -> Reader

Returns a Reader over the tree root, to pass to read.

A struct is read from a Map by key, a slice from a Seq, an optional from Null, and an enum variant from a Text of its name or a Map with one member named by the variant. An integer is read from Int, and from Uint only as a u64; a u64 is read from a non-negative Int as well. A float is read from Float or an integer. A []u8 is read from Binary, and from Text holding base64 when base64_text is true. Keys are matched as key_text gives. Fails with Mismatch when a value has another type and with Range when a number does not fit. position is the at of the node last read, or the number of nodes read so far when a node has at of 0.

The source that a value is read from, made of closures that a format implements.

read calls the closures in document order and each reading closure consumes one value. Reading closures fail with Mismatch when the next value has another type, Range when a number does not fit, and Syntax when the input has no value to read.

read

pub fn read[T](r: Reader) -> T ! Error

Reads a T from r.

A struct is opened with begin_map and read through next_field: unknown fields are skipped, a missing field takes the field's default or none for an optional field, and any other missing field fails with Missing. Integer types narrower than i64 are range checked and fail with Range. A slice is read as a sequence. A map is read through next_key and its keys are parsed from the key text, failing with Mismatch when an integer or boolean key is not one and with Range when it does not fit. An enum is read from the variant name that begin_variant returns, failing with UnknownVariant for an unknown name; a labelled variant reads its fields like a struct, and a positional variant fails with Mismatch when the number of values is wrong. Other failures come from r.

The sink that a value is written to, made of closures that a format implements.

write calls the closures in document order. A sequence, map, struct or variant is opened by a begin_ closure and closed by the matching end_ closure, with its elements written between them by the closures for their own types. A map entry is key followed by its value, and a struct field is field followed by its value.

node_of

pub fn node_of[V](v: V, shape: fn(V) -> Shape[V]) -> Node

Returns the Node for the value v of a format's own type, where shape says whether v is a leaf, a sequence or a map.

The node and the nodes below it carry the at that shape gives.

parse_integer

pub fn parse_integer(s: str, what: str, at: i64) -> i64 ! Error

Parses s as a signed decimal integer.

A single leading - or + is accepted. Fails with Mismatch naming what when s is not an integer, and with Range when it does not fit an i64, both at position at.

parse_unsigned

pub fn parse_unsigned(s: str, what: str, at: i64) -> u64 ! Error

Parses s as an unsigned decimal integer.

Fails with Mismatch naming what when s has anything but digits, and with Range when it does not fit a u64, both at position at.

syntax_at

to_tree

pub fn to_tree[T](v: T) -> Node

Returns v as a Tree, through write.

A struct and a map become a Map with Text keys, a slice a Seq, a []u8 a Binary, a none optional Null, and an enum variant without fields a Text of its name. A variant with fields becomes a Map with one member named by the variant, whose value is a Map of its labelled fields or a Seq of its positional ones. Every node has at of 0.

unsigned_tree

pub fn unsigned_tree(n: u64) -> Tree

Returns n as a Tree: Int when it fits an i64 and Uint above 9223372036854775807.

write

pub fn write[T](v: T, w: Writer)

Writes v to w.

The compiler derives the code for the concrete type T: booleans, integers, floats and strings map to the Writer primitives, []u8 to binary, slices to sequences, maps to maps with keys written as text (keys must be strings, integers or booleans), optionals to null or the value, and pointers to their target, a shared pointer in full each time. A pointer reached again while its target is being written is a cycle and panics with serde: cycle through *T. A struct is written with every field, in declaration order. An enum variant without fields is written with unit_variant; one with fields with begin_variant, followed by a struct of its fields when they are labelled and a sequence otherwise. A type that cannot be derived is a compile error.
