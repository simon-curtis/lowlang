std/term

#os

import std/term

is_tty, size and set_size. enter_raw returns a droppable Raw that restores the terminal when released. Key, read_key (a lone escape resolves after 50 ms) and parse_keys. ANSI helpers: fg, bg with 16, 256 and RGB colours, bold, dim, underline, cursor movement, clear_screen, the alternate screen, strip and visible_width. colour_enabled honours NO_COLOR, CLICOLOR_FORCE and TERM=dumb. open_pty.

Imports std/atomic, std/io, std/os, std/sys, std/text.

Examples

menu

import std/term
import std/text

fn label(k: term.Key) -> str {
    match k {
        Char(c) => "char {text.from_char(c)}"
        Ctrl(c) => "ctrl+{text.from_char(c)}"
        Alt(c) => "alt+{text.from_char(c)}"
        Function(n) => "F{n}"
        _ => "{k}"
    }
}

/// Decodes recorded terminal input into key presses and drives a menu cursor with them.
fn main() {
    let esc = text.from_char(27)
    let input = "{esc}[B{esc}[B{esc}[A\rj{esc}x{text.from_char(3)}{esc}OP{esc}[3~é"
    let keys = term.parse_keys(bytes(input))
    let items = ["open", "save", "quit"]
    var selected = 0
    for k in keys {
        match k {
            Down => {
                selected = (selected + 1) % items.len
            }
            Up => {
                selected = (selected + items.len - 1) % items.len
            }
            Enter => print("chose {items[selected]}")
            _ => {}
        }
        print("{label(k)}, cursor on {items[selected]}")
    }
    print("truncated sequence: {term.parse_keys(bytes("{esc}["))}")
}

pty

import std/io
import std/sys
import std/term
import std/text

fn describe(k: term.Key) -> str {
    match k {
        Char(c) => "char {text.from_char(c)}"
        Ctrl(c) => "ctrl+{text.from_char(c)}"
        _ => "{k}"
    }
}

/// Opens a pseudoterminal, sizes it, switches its slave to raw mode, types into it from the master side and reads the key presses back.
fn main() ! io.Error {
    using pty = term.open_pty()?
    let master = pty.handle.master
    let slave = pty.handle.slave
    print("master is a terminal: {term.is_tty(master)}")

    term.set_size(slave, term.Size { columns: 100, rows: 30 })?
    let size = term.size(slave)?
    print("size: {size.columns}x{size.rows}")

    print("raw before: {term.is_raw(slave)?}")
    using raw = term.enter_raw(slave)?
    print("raw during: {term.is_raw(slave)?}")

    let typed = "a{text.from_char(27)}[A{text.from_char(3)}"
    sys.write_all(master, bytes(typed))?
    for _ in 0..3 {
        let key = term.read_key(slave)?
        print("key: {describe(key)}")
    }

    term.restore(raw)
    print("raw after: {term.is_raw(slave)?}")
}

styling

import std/term
import std/text

struct Check {
    name: str,
    ok: bool,
}

fn show(s: str) -> str {
    text.replace(s, text.from_char(27), "<ESC>")
}

fn badge(ok: bool) -> str {
    if ok {
        term.fg(term.Colour.Green, "pass")
    } else {
        term.bold(term.fg(term.Colour.Red, "FAIL"))
    }
}

fn pad(s: str, width: i64) -> str {
    let missing = width - term.visible_width(s)
    "{s}{text.repeat(" ", missing)}"
}

/// Builds coloured status lines, aligns them by their visible width, shows the raw escape sequences, and strips them again for plain output.
fn main() {
    let rows = [
        Check { name: "lexer", ok: true },
        Check { name: "parser", ok: true },
        Check { name: "checker", ok: false },
    ]
    var lines: []str = []
    for row in rows {
        let name = term.underline(row.name)
        push(mut lines, "{pad(name, 10)}{badge(row.ok)}")
    }

    for line in lines {
        print("raw:   {show(line)}")
        print("plain: {term.strip(line)}")
        print("width: {term.visible_width(line)} of {line.len} bytes")
    }

    print(show(term.fg(term.Colour.Indexed(208), "orange")))
    print(show(term.bg(term.Colour.Rgb(10, 20, 30), "dark")))
    print(show("{term.hide_cursor()}{term.cursor_to(3, 7)}{term.clear_line()}{term.show_cursor()}"))
}

pub fn enter_raw(fd: i64) -> Raw ! io.Error

Puts the terminal on fd in raw mode and returns the owner that restores it.

Raw mode turns off line buffering, echo, signal-generating keys, flow control and input and output translation, uses 8-bit characters, and makes reads return after one byte. Fails with Unsupported if fd is not a terminal.

parse_keys

pub fn parse_keys(data: []u8) -> []Key

Decodes the key presses in data, which holds bytes already read from a terminal.

Does not wait: a sequence cut short by the end of data gives Unknown, and a lone escape byte at the end gives Escape. Never fails.

read_key

pub fn read_key(fd: i64) -> Key ! io.Error

Reads one key press from the descriptor fd, waiting 50 milliseconds after an escape byte to tell an Escape key from a longer sequence.

Does the same as read_key_with with that timeout.

read_key_with

pub fn read_key_with(fd: i64, escape_ms: i64) -> Key ! io.Error

Reads one key press from the descriptor fd.

Parks the green thread until the first byte arrives. After an escape byte, and while it reads the rest of a sequence, it waits at most escape_ms milliseconds for each following byte and reports Escape when none comes. A multi-byte UTF-8 character is decoded to Char, and a malformed one gives Char(0xFFFD). The terminal should be in raw mode, or input arrives only after Enter. Fails with Eof at end of input and with the system error otherwise.
