std/json/schema

#formats

import std/json/schema

JSON Schema 2020-12. compile and compile_with check the schema, resolve $ref, $anchor, $dynamicRef and registered resources, and precompile patterns. validate returns Violations with the instance pointer, the keyword pointer and a message. unevaluatedProperties and unevaluatedItems use annotations from the branches that passed. format annotates or asserts.

Imports std/json, std/json/pointer, std/regex, std/slices, std/text, std/uri.

Examples

validate

import std/json
import std/json/schema
import std/text

fn parse(source: str) -> json.Value {
    json.parse(bytes(text.replace(source, "'", "\""))) or json.Value.Null
}

fn check(s: schema.Schema, label: str, source: str) {
    let violations = schema.validate(s, parse(source))
    if violations.len == 0 {
        print("{label}: valid")
        return
    }
    print("{label}: violations {violations.len}")
    for v in violations {
        let where = if v.instance == "" { "(root)" } else { v.instance }
        print("  {where} {v.keyword}: {v.message}")
    }
}

/// Compiles a schema for user records and reports every violation of each submitted record.
fn main() {
    let definition = parse("{{'type': 'object', 'required': ['name', 'age'], 'properties': {{'name': {{'type': 'string', 'minLength': 1}}, 'age': {{'type': 'integer', 'minimum': 0, 'maximum': 150}}, 'email': {{'type': 'string', 'format': 'email'}}, 'roles': {{'type': 'array', 'items': {{'enum': ['admin', 'editor', 'viewer']}}, 'uniqueItems': true}}}}, 'additionalProperties': false}}")
    let s = schema.compile(definition) catch {
        Malformed(at, message) => {
            print("schema is malformed at {at}: {message}")
            return
        }
        Unresolved(reference) => {
            print("unresolved reference {reference}")
            return
        }
    }

    check(s, "valid", "{{'name': 'ada', 'age': 36, 'roles': ['admin']}}")
    check(s, "missing", "{{'name': 'bob'}}")
    check(s, "ranges", "{{'name': '', 'age': 200}}")
    check(s, "roles", "{{'name': 'eve', 'age': 5, 'roles': ['admin', 'root', 'admin']}}")
    check(s, "extra", "{{'name': 'x', 'age': 1, 'admin': true}}")
    check(s, "not an object", "[1, 2]")

    let strict = schema.compile_with(definition, schema.Options { formats: Assert }) or return
    print("lenient email: {schema.is_valid(s, parse("{{'name': 'a', 'age': 1, 'email': 'nobody'}}"))}")
    print("strict email: {schema.is_valid(strict, parse("{{'name': 'a', 'age': 1, 'email': 'nobody'}}"))}")

    let broken = schema.compile(parse("{{'type': 'integr'}}")) catch {
        Malformed(at, message) => {
            print("schema is malformed at {at}: {message}")
            return
        }
        Unresolved(reference) => {
            print("unresolved reference {reference}")
            return
        }
    }
    print("compiled {schema.is_valid(broken, parse("1"))}")
}

Schema

pub struct Schema {

A compiled JSON Schema (draft 2020-12), made by compile or compile_with.

A Schema can validate any number of instances.

is_valid

pub fn is_valid(s: Schema, instance: json.Value) -> bool

Returns whether instance is valid against s; equivalent to validate(s, instance).len == 0.

validate

pub fn validate(s: Schema, instance: json.Value) -> []Violation

Returns every way in which instance fails s; the result is empty when it is valid.

Annotations from the branches of allOf, anyOf, oneOf, if and the other applicators that passed decide what unevaluatedProperties and unevaluatedItems cover. Validation does not stop at the first failure.

compile_with

pub fn compile_with(schema: json.Value, options: Options) -> Schema ! Error

Compiles the JSON Schema 2020-12 document schema.

It checks the schema, resolves $ref, $anchor, $dynamicRef and the schemas in options.resources by their $id, and precompiles every pattern and patternProperties expression. A schema is an object or a boolean. Fails with Malformed for an invalid schema and Unresolved for a reference that resolves to nothing.
