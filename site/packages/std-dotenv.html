std/dotenv

#formats

import std/dotenv

.env files. parse and read return a map from NAME=value lines. They allow export, blank lines, # comments (inline after whitespace in unquoted values), single quotes taken literally, and double quotes with \n, \t and \r escapes that may span lines. Errors carry the line.

Imports std/file, std/io, std/text.

Examples

config

import std/dotenv
import std/text

struct Config {
    host: str
    port: i64
    debug: bool
    banner: str
}

fn settings(source: str) -> Config ! dotenv.Error {
    let vars = dotenv.parse(source)?
    Config {
        host: vars["HOST"] ?? "localhost",
        port: text.parse_int(vars["PORT"] ?? "") or 8080,
        debug: text.parse_bool(vars["DEBUG"] ?? "") or false,
        banner: vars["BANNER"] ?? "",
    }
}

fn describe(e: dotenv.Error) -> str {
    match e {
        Read(cause) => "unreadable"
        MissingEquals(line) => "line {line} has no ="
        BadName(line) => "line {line} has a bad name"
        UnterminatedQuote(line) => "quote opened on line {line} is never closed"
    }
}

fn show(label: str, source: str) {
    let config = settings(source) catch {
        e => {
            print("{label}: {describe(e)}")
            return
        }
    }
    print("{label}: {config.host}:{config.port} debug={config.debug}")
    for line in text.lines(config.banner) {
        print("  | {line}")
    }
}

/// Loads typed settings from `.env` text, falling back to defaults for missing or invalid values.
fn main() {
    show("full", "# service settings\nexport HOST=api.internal\nPORT=9090 # listen port\nDEBUG=true\nBANNER=\"Welcome\\nto the API\"\n")
    show("defaults", "HOST='raw $value'\nPORT=not-a-number\n")
    show("missing equals", "HOST=a\nPORT\n")
    show("bad name", "9HOST=a\n")
    show("open quote", "BANNER=\"never closed\nPORT=1\n")
}

Output

  | to the API

Functions

parse

pub fn parse(src: str) -> [str: str] ! Error

Parses .env text into a map from variable name to value.

Each line is blank, a # comment, or NAME=value with an optional leading export . Whitespace around the name and before the value is ignored. An unquoted value runs to the end of the line, stops at a # that starts the value or follows a space or tab, and is trimmed. A value in single quotes is taken literally. A value in double quotes processes \n, \t and \r; any other backslash pair becomes the character after the backslash. Quoted values may span lines, and text after the closing quote on its line is ignored. A name defined twice keeps its last value. Variables are not expanded.

Fails with MissingEquals, BadName or UnterminatedQuote, each carrying a line number.

read

pub fn read(path: str) -> [str: str] ! Error

Reads the file at path and parses it as .env text.

Fails with Read when the file cannot be read as text, and otherwise with the errors of parse.
