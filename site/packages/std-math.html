std/math

#math

import std/math

libm's trigonometric, hyperbolic, exponential and rounding functions, pi, tau, e, infinity, nan, is_nan, is_inf, generic abs, min, max, clamp and compare, and integer ipow, gcd, floor_div and floor_mod.

Examples

geometry

import std/math

struct Point {
    x: i64
    y: i64
}

fn on_circle(radius: f64, degrees: i64) -> Point {
    let angle = f64(degrees) * math.pi / 180.0
    Point { x: i64(math.round(radius * math.cos(angle))), y: i64(math.round(radius * math.sin(angle))) }
}

/// Lays out a ring of markers, measures distances between them and wraps indexes and offsets around.
fn main() {
    let width = 1920
    let height = 1080
    let divisor = math.gcd(width, height)
    print("{width}x{height} is {width / divisor}:{height / divisor}")

    var ring: []Point = []
    for step in 0..8 {
        push(mut ring, on_circle(10.0, step * 45))
    }
    for i, p in ring {
        print("marker {i} at ({p.x}, {p.y})")
    }

    let a = ring[0]
    let b = ring[3]
    let distance = math.hypot(f64(b.x - a.x), f64(b.y - a.y))
    print("marker 0 to marker 3: {math.round(distance * 100.0) / 100.0}")
    print("angle of marker 2 from the origin: {math.round(math.atan2(f64(ring[2].y), f64(ring[2].x)) / math.pi * 180.0)} degrees")

    print("previous marker of 0: {math.floor_mod(0 - 1, ring.len)}, marker 11 wraps to {math.floor_mod(11, ring.len)}")
    print("minutes -90 are {math.floor_div(-90, 60)} hours and {math.floor_mod(-90, 60)} minutes before the hour")

    let offsets = [-250, -30, 0, 45, 400]
    var clamped: []i64 = []
    for offset in offsets {
        push(mut clamped, math.clamp(offset, -100, 100))
    }
    print("clamped offsets {clamped}")
    print("largest distance from zero: {math.max(math.abs(offsets[0]), math.abs(offsets[4]))}")
    print("compare {math.compare(3, 5)} {math.compare(5, 5)} {math.compare(7, 5)}")
    print("2^10 = {math.ipow(2, 10)}, as a float {math.pow(2.0, 10.0)}, is nan: {math.is_nan(math.nan)}")
}

Functions

abs

pub fn abs[T: Number](x: T) -> T

Returns the absolute value of x, which is -x when x is less than zero and x otherwise.

The negation has the overflow behaviour of T, so the minimum value of a signed integer type stays negative.

acos

pub extern fn acos(x: f64) -> f64

Returns the arc cosine of x in radians, in the range 0 to pi.

Returns nan when x is outside -1 to 1.

asin

pub extern fn asin(x: f64) -> f64

Returns the arc sine of x in radians, in the range -pi/2 to pi/2.

Returns nan when x is outside -1 to 1.

atan

pub extern fn atan(x: f64) -> f64

Returns the arc tangent of x in radians, in the range -pi/2 to pi/2.

atan2

pub extern fn atan2(y: f64, x: f64) -> f64

Returns the angle in radians, in the range -pi to pi, of the point at x along the horizontal axis and y along the vertical axis.

The signs of both arguments choose the quadrant.

cbrt

pub extern fn cbrt(x: f64) -> f64

Returns the real cube root of x, which is negative for a negative x.

ceil

pub extern fn ceil(x: f64) -> f64

Returns the smallest whole number that is at least x.

clamp

pub fn clamp[T: Ord](x: T, lo: T, hi: T) -> T

Returns x limited to the range lo to hi.

Returns lo when x is below it and hi when x is above it. lo must not be greater than hi.

compare

pub fn compare[T: Ord](a: T, b: T) -> i64

Returns -1, 0 or 1 when a is less than, equal to or greater than b.

cos

pub extern fn cos(x: f64) -> f64

Returns the cosine of x, an angle in radians.

cosh

pub extern fn cosh(x: f64) -> f64

Returns the hyperbolic cosine of x.

floor_div

pub fn floor_div(a: i64, b: i64) -> i64

Returns a divided by b, rounded toward negative infinity.

Panics when b is zero.

floor_mod

pub fn floor_mod(a: i64, b: i64) -> i64

Returns the remainder of a divided by b with the sign of b, so that floor_div(a, b) * b + floor_mod(a, b) is a.

Panics when b is zero.

fmod

pub extern fn fmod(x: f64, y: f64) -> f64

Returns the remainder of x divided by y, with the sign of x.

The quotient is truncated toward zero, so the result is x - n * y for the integer n that gives a result smaller in magnitude than y.

gcd

pub fn gcd(a: i64, b: i64) -> i64

Returns the greatest common divisor of a and b, which is never negative.

The signs of the arguments are ignored and gcd(0, 0) is 0.

hypot

pub extern fn hypot(x: f64, y: f64) -> f64

Returns the length of the hypotenuse of a right triangle with legs x and y, which is sqrt(x * x + y * y) without intermediate overflow.

ipow

pub fn ipow(base: i64, exp: i64) -> i64

Returns base raised to the integer power exp, computed by repeated squaring.

Returns 1 when exp is zero or negative. Overflow wraps like i64 multiplication.

is_inf

pub fn is_inf(x: f64) -> bool

Returns true when x is positive or negative infinity.

is_nan

pub fn is_nan(x: f64) -> bool

Returns true when x is not a number.

nan is the only value that is not equal to itself, so this is the way to test for it.

log

pub extern fn log(x: f64) -> f64

Returns the natural logarithm of x.

Returns negative infinity for zero and nan for a negative x.

log10

pub extern fn log10(x: f64) -> f64

Returns the base 10 logarithm of x.

Returns negative infinity for zero and nan for a negative x.

log2

pub extern fn log2(x: f64) -> f64

Returns the base 2 logarithm of x.

Returns negative infinity for zero and nan for a negative x.

max

pub fn max[T: Ord](a: T, b: T) -> T

Returns the larger of a and b.

Returns a when they are equal. When either is not a number, returns a.

min

pub fn min[T: Ord](a: T, b: T) -> T

Returns the smaller of a and b.

Returns a when they are equal. When either is not a number, returns a.

pow

pub extern fn pow(x: f64, y: f64) -> f64

Returns x raised to the power y.

round

pub extern fn round(x: f64) -> f64

Returns the whole number nearest to x, with halfway cases rounded away from zero.

Returns the hyperbolic sine of x.

tan

Returns the tangent of x, an angle in radians.

tanh

pub extern fn tanh(x: f64) -> f64

Returns the hyperbolic tangent of x.

trunc

pub extern fn trunc(x: f64) -> f64

Returns x with its fractional part removed, which rounds toward zero.
