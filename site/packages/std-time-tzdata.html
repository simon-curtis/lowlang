std/time/tzdata

#time

import std/time/tzdata

An embedded copy of the database for hosts without one: source (for zone.load_from), names and version. The data is linked only when source or names is reachable.

Imports std/text.

Examples

zone_picker

import std/text
import std/time/fault
import std/time/offset
import std/time/timestamp
import std/time/tzdata
import std/time/zone

fn suggestions(typed: str) -> []str {
    var found: []str = []
    for name in tzdata.names() {
        if text.starts_with_fold(name, typed) {
            push(mut found, name)
        }
    }
    found
}

/// Offers zone names for what a user typed and reads the chosen zone from the embedded database, with no files on the host.
fn main() ! fault.Fault {
    print("database version starts with a year: {tzdata.version()[0] == '2'}")
    print("zones embedded: {tzdata.names().len > 300}")

    for typed in ["europe/lo", "Pacific/Au", "Antarctica/Mc", "Narnia"] {
        let options = suggestions(typed)
        print("{typed}: {options}")
    }

    let source = tzdata.source()
    let raw = source("America/Sao_Paulo") ?? []
    print("stored data is TZif: {text.starts_with(str(raw), "TZif")}")
    let unknown = source("Narnia/Cair_Paravel") ?? []
    print("unknown name has no data: {unknown.len == 0}")

    let sao_paulo = zone.decode("America/Sao_Paulo", raw)?
    let moment = timestamp.parse("2026-11-13T15:00:00Z")?
    let info = zone.info_at(sao_paulo, moment)
    print("{zone.name(sao_paulo)} is {offset.to_string(info.offset)}, no daylight saving: {!info.dst}")

    let same = zone.load_from("America/Sao_Paulo", source)?
    print("load_from gives the same offset: {offset.equal(zone.offset_at(same, moment), info.offset)}")
}

Functions

names

pub fn names() -> []str

Returns the names of all embedded zones in index order.

source

pub fn source() -> fn(str) -> []u8?

Returns a function that gives the embedded TZif data for a zone name, or none for an unknown name.

It is the source argument for zone.load_from.

version

pub fn version() -> str

Returns the version label of the IANA time zone database embedded in the program.
