std/base64

#encoding

import std/base64

One encoder, encode_with(data, alphabet, padded), and one decoder, decode_with(s, alphabet, padding), over an alphabet (standard, url_safe) and a Padding rule (Required, Forbidden, Optional). encode, encode_unpadded and encode_url, and decode, decode_unpadded and decode_url, fix the alphabet and padding. Every decoder rejects non-zero trailing bits and a byte outside the alphabet.

Examples

token

import std/base64
import std/text

fn segment(label: str, s: str) {
    let data = base64.decode_url(s) catch {
        BadLength => {
            print("{label}: bad length")
            return
        }
        BadByte(at) => {
            print("{label}: bad byte at {at}")
            return
        }
    }
    print("{label}: {str(data)}")
}

/// Builds an unsigned token from URL-safe segments, reads the segments back and rejects malformed ones.
fn main() {
    let header = base64.encode_url(bytes("{{\"alg\":\"none\",\"typ\":\"JWT\"}}"))
    let payload = base64.encode_url(bytes("{{\"sub\":\"user?>1\"}}"))
    let token = "{header}.{payload}."
    print(token)

    let parts = text.split(token, '.')
    segment("header", parts[0])
    segment("payload", parts[1])

    segment("standard alphabet", "dXNlcj8+MQ")
    segment("truncated", "dXNlcj8+M")

    let padded = base64.encode(bytes("user?>1"))
    print(padded)
    let bare = padded[..padded.len - 2]
    for form in [padded, bare] {
        let data = base64.decode_with(form, base64.standard, Optional) or bytes("")
        print("optional padding {form}: {str(data)}")
    }
}

Functions

decode

pub fn decode(s: str) -> []u8 ! DecodeError

Decodes padded standard Base64. The length must be a multiple of 4, so unpadded input fails with BadLength. = is accepted only as the last one or two characters. Fails as decode_with does.

decode_unpadded

pub fn decode_unpadded(s: str) -> []u8 ! DecodeError

Decodes unpadded standard Base64. = is rejected. Fails as decode_with does.

decode_url

pub fn decode_url(s: str) -> []u8 ! DecodeError

Decodes unpadded URL-safe Base64. =, + and / are rejected. Fails as decode_with does.

decode_with

pub fn decode_with(s: str, alphabet: str, padding: Padding) -> []u8 ! DecodeError

Decodes s written with the 64 characters of alphabet and the given padding. Fails with BadLength when the length does not suit the padding, and with BadByte at the index of the first byte outside the alphabet, of a misplaced or disallowed =, or, at the final index, when the bits left over in the last character are not zero. Empty input gives empty data.

encode

pub fn encode(data: []u8) -> str

Returns data encoded as standard Base64 (RFC 4648 section 4), padded with = to a multiple of 4 characters.

encode_unpadded

pub fn encode_unpadded(data: []u8) -> str

Returns data encoded as standard Base64 without padding.

encode_url

pub fn encode_url(data: []u8) -> str

Returns data encoded as URL-safe Base64 (RFC 4648 section 5) without padding. The alphabet uses - and _ in place of + and /.

encode_with

pub fn encode_with(data: []u8, alphabet: str, padded: bool) -> str

Returns data encoded with the 64 characters of alphabet, padded with = to a multiple of 4 characters when padded is true.
