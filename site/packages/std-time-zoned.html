std/time/zoned

#time

import std/time/zoned

Zoned, a Timestamp seen through a Zone whose offset is always the zone's own. Built from a civil time with a Disambiguation (Compatible, Earlier, Later, Reject) or strict. plus by Span moves calendar units on civil dates and time units by elapsed time; plus_duration is exact. Also until and since with a Diff, start_of_day, hours_in_day, series and round. RFC 9557 to_string (2026-11-11T10:00:00+00:00[Europe/London]), to_rfc3339, to_rfc2822, and parse with an OffsetPolicy for a written offset that disagrees with the zone. Pattern format and parse_with.

Imports std/text, std/time/clock, std/time/date, std/time/datetime, std/time/duration, std/time/fault, std/time/offset, std/time/pattern, std/time/rounding, std/time/span, std/time/timeofday, std/time/timestamp, std/time/zone.

Examples

ingest

import std/time/fault
import std/time/rounding
import std/time/span
import std/time/timestamp
import std/time/tzdata
import std/time/zone
import std/time/zoned

fn load_zone(name: str) -> zone.Zone ! fault.Fault {
    zone.load_from(name, tzdata.source())?
}

fn ingest(line: str, policy: zoned.OffsetPolicy) -> str {
    let z = zoned.parse_using_options(line, policy, load_zone) catch {
        OffsetConflict => return "offset disagrees with the zone"
        UnknownZone(name) => return "unknown zone {name}"
        Syntax(at) => return "syntax error at byte {at}"
        f => return fault.to_string(f)
    }
    "{zoned.to_rfc3339(z)} {zoned.abbreviation(z)}"
}

/// Reads RFC 9557 timestamps from other systems, choosing what to trust when the written offset and the zone disagree.
fn main() ! fault.Fault {
    let lines = [
        "2026-07-01T12:00:00+01:00[Europe/London]",
        "2026-07-01T12:00:00Z[Europe/London]",
        "2026-07-01T12:00:00[Europe/London]",
        "2026-07-01T12:00:00+05:00[Europe/London]",
        "2026-07-01T12:00:00+05:30[+05:30]",
        "2026-07-01T12:00:00[Atlantis/Lost]",
        "2026-07-01T12:00:00+01:00[Europe/London][u-ca=hebrew]",
        "2026-07-01 12:00"
    ]
    for line in lines {
        print("{line}")
        print("  {ingest(line, zoned.OffsetPolicy.Reject)}")
    }

    let conflict = "2026-07-01T12:00:00+05:00[Europe/London]"
    print("same conflict under each policy")
    print("  prefer_offset {ingest(conflict, zoned.OffsetPolicy.PreferOffset)}")
    print("  use_offset    {ingest(conflict, zoned.OffsetPolicy.UseOffset)}")
    print("  use_zone      {ingest(conflict, zoned.OffsetPolicy.UseZone)}")

    let sent = zoned.parse_using_options(conflict, zoned.OffsetPolicy.UseOffset, load_zone)?
    let received = zoned.parse_using_options("2026-07-01T07:00:00Z", zoned.OffsetPolicy.Reject, load_zone)?
    print("same instant {timestamp.equal(zoned.timestamp(sent), zoned.timestamp(received))}")
    let rounded = zoned.round(zoned.parse_using("2026-07-01T12:29:31+01:00[Europe/London]", load_zone)?, span.Unit.Minutes, 30, rounding.Rounding.HalfExpand)?
    print(zoned.to_string(rounded))
}

meeting

import std/time/datetime
import std/time/duration
import std/time/fault
import std/time/span
import std/time/tzdata
import std/time/zone
import std/time/zoned

fn load_zone(name: str) -> zone.Zone ! fault.Fault {
    zone.load_from(name, tzdata.source())?
}

fn show(label: str, z: zoned.Zoned) {
    print("{label}: {zoned.format(z, "%a %d %b %H:%M %Z (%:z)") or "?"}")
}

/// Schedules a weekly meeting in London, shows it in other cities, and follows it across both daylight saving changes.
fn main() ! fault.Fault {
    let london = load_zone("Europe/London")?
    let new_york = load_zone("America/New_York")?
    let tokyo = load_zone("Asia/Tokyo")?

    let first = zoned.strict(datetime.of(2026, 10, 15, 16, 0, 0, 0)?, london)?
    show("London", first)
    show("New York", zoned.in_zone(first, new_york)?)
    show("Tokyo", zoned.in_zone(first, tokyo)?)

    let weekly = zoned.series(first, span.weeks(1), 5, zoned.Disambiguation.Compatible)?
    print("weekly in London, with the New York wall clock beside it")
    for meeting in weekly {
        let there = zoned.in_zone(meeting, new_york)?
        print("  {zoned.format(meeting, "%d %b %H:%M %Z")?}  =  {zoned.format(there, "%H:%M %Z")?}")
    }

    let gap = datetime.of(2026, 3, 29, 1, 30, 0, 0)?
    let moved = zoned.from_civil(gap, london, zoned.Disambiguation.Compatible)?
    show("01:30 in the spring gap, Compatible", moved)
    let earlier = zoned.from_civil(gap, london, zoned.Disambiguation.Earlier)?
    show("01:30 in the spring gap, Earlier", earlier)
    zoned.strict(gap, london) catch {
        Skipped => {
            print("strict refuses the skipped time")
            moved
        }
        _ => moved
    }

    let eve = zoned.strict(datetime.of(2026, 3, 28, 9, 0, 0, 0)?, london)?
    let by_calendar = zoned.plus(eve, span.days(1), zoned.Disambiguation.Compatible)?
    let by_elapsed = zoned.plus_duration(eve, duration.hours(24))?
    show("a day later by calendar", by_calendar)
    show("24 hours later", by_elapsed)

    let change_day = zoned.start_of_day(zoned.strict(datetime.of(2026, 10, 25, 12, 0, 0, 0)?, london)?)?
    print("the day the clocks go back is {duration.friendly(zoned.hours_in_day(change_day)?)} long")
    print("the day the clocks go forward is {duration.friendly(zoned.hours_in_day(by_calendar)?)} long")

    let next = zoned.next_transition(first)
    match next {
        some(change) => print("next London change {zoned.to_string(zoned.of(change.at, london)?)}")
        none => print("no further changes")
    }

    let wire = zoned.to_string(first)
    print(wire)
    let back = zoned.parse_using(wire, load_zone)?
    print("parsed back equal {zoned.equal(back, first)}, rfc 2822: {zoned.to_rfc2822(first)}")

    let spread = zoned.until(first, weekly[4], span.Diff { largest: span.Unit.Weeks })?
    print("first to last meeting {span.friendly(spread)}, {duration.friendly(zoned.duration_until(first, weekly[4]))} elapsed")
}

Disambiguation

pub enum Disambiguation {Takes the first occurrence in an overlap and, in a gap, moves the time forward by the length of the gap. A local 02:30 that does not exist becomes 03:30.    CompatibleTakes the first occurrence in an overlap and, in a gap, moves the time back by the length of the gap. A local 02:30 that does not exist becomes 01:30.    EarlierTakes the second occurrence in an overlap and, in a gap, moves the time forward as Compatible does.    LaterFails with Ambiguous in an overlap and with Skipped in a gap.    Reject
}

How a local date-time that a zone's daylight saving change makes ambiguous or nonexistent is turned into an instant.

A local time occurs twice when the clocks go back, an overlap, and not at all when they go forward, a gap.

OffsetPolicy

pub enum OffsetPolicy {Uses a valid offset and fails with OffsetConflict for any other.    RejectUses a valid offset, and for any other resolves the local time by the zone's rules, as Disambiguation.Compatible does.    PreferOffsetAlways uses the written offset. The instant is the one the offset gives, shown in the zone, so the local time can differ from the one written.    UseOffsetIgnores the written offset and resolves the local time by the zone's rules, as Disambiguation.Compatible does.    UseZone
}

What parse and its variants do with an offset written next to a zone.

An offset is valid when the zone has it at the written local time.

Zoned

pub struct Zoned {
    ...
}

An instant in a time zone: the moment, the zone, and the local date-time and offset the zone has at that moment.

It is the type for human-facing times. Arithmetic on the calendar fields follows the zone's wall clock, so adding a day keeps the local time across a daylight saving change, while adding a duration adds exact elapsed time. Comparison looks at the instant only. Copies share the zone's rules.

pub fn plus(z: Zoned, s: span.Span, how: Disambiguation) -> Zoned ! fault.Fault

round

pub fn round(z: Zoned, unit: span.Unit, increment: i64, mode: rounding.Rounding) -> Zoned ! fault.Fault

Returns z rounded to a multiple of increment units of local time using mode.

unit can be Days, which needs an increment of 1 and rounds to the start of this or the next day by the actual length of the day. A smaller unit rounds the local wall time. When the result is in an overlap it takes the later occurrence only if z is already on it, and when it is in a gap it uses the offset before the change. Fails with Invalid when unit is a larger calendar unit or the increment does not divide the next larger unit as timeofday.rounded_nanos states, and with OutOfRange or Overflow when the result is outside the supported years.

until

pub fn until(a: Zoned, b: Zoned, diff: span.Diff) -> span.Span ! fault.Fault

Returns the span from a to b, measured and rounded as diff says.

When diff.largest is a calendar unit, b is moved into the zone of a and years, months, weeks and days are counted on its wall clock, so a day can be 23 or 25 hours long, and the remainder is split into time units. Otherwise the span is the exact elapsed time split into time units from largest down. Rounding is relative to a. The span is negative when b is earlier. Fails with Invalid when the options are inconsistent as span.check_diff and span.check_time_increment state, and with OutOfRange or Overflow where the result cannot be represented.

parse_names

pub fn parse_names(s: str, p: pattern.Pattern, names: pattern.Names) -> Zoned ! fault.Fault

Parses s with the compiled pattern p and the words of names into a time in a fixed-offset zone.

The pattern must either give the Unix time with %s, or give a full date, a time of day and an offset with %z. A Z or z is read as a zero offset. The zone is a fixed zone of the written offset, or UTC when only %s is given. Other fields that are present must agree with the instant. Fails with Invalid when the fields do not give an instant or conflict, with Syntax or OutOfRange when s does not match, and with OutOfRange for an offset that is too large.

parse_names_zone

pub fn parse_names_zone(s: str, p: pattern.Pattern, names: pattern.Names, tz: zone.Zone, how: Disambiguation) -> Zoned ! fault.Fault

Parses s with the compiled pattern p and the words of names into a time in the zone tz, with how deciding what happens when the local time is an overlap or a gap.

The pattern gives either a date and a time of day or the Unix time with %s. An offset written with %z must be one that tz has at that local time or the parse fails with OffsetConflict. A zone abbreviation written with %Z must match the one the zone has at that instant or the parse fails with Invalid. Also fails as from_civil does.

parse_using_options

pub fn parse_using_options(s: str, policy: OffsetPolicy, loader: fn(str) -> zone.Zone ! fault.Fault) -> Zoned ! fault.Fault

Parses a zoned date-time such as 2024-03-01T12:30:00+01:00[Europe/Paris], with policy deciding what happens when the offset disagrees with the zone and loader supplying the zone for a name.

The text is a date-time as datetime.parse reads it, then an optional Z or offset, then bracketed annotations. A name in brackets such as [Europe/Paris] is passed to loader, and [+01:00] is a fixed offset. A [u-ca=iso8601] calendar annotation is accepted, and another calendar fails with Unsupported. A ! in front of an annotation marks it critical, and an unknown critical annotation fails with Unsupported. Unknown annotations that are not critical are ignored. Without a zone annotation the offset or Z is required and the result has a fixed-offset zone. With a zone annotation the offset is optional, and without it the local time is resolved as Disambiguation.Compatible does. Fails with Syntax at the offset of the first bad byte, with OffsetConflict under Reject, and with the fault loader returns for an unknown name.
