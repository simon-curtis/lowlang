RFC 9535 JSONPath: compile, query, find, values and paths, with slices, filters, match, search, length, count and value, normalised paths, and syntax errors at a byte offset.

Imports std/json, std/math, std/regex, std/text, std/unicode/utf16.

Examples

catalog

import std/json
import std/json/path
import std/text

fn parse(source: str) -> json.Value {
    json.parse(bytes(text.replace(source, "'", "\""))) or json.Value.Null
}

fn show(nodes: []path.Node) -> str {
    text.join(path.values(nodes) |> texts(), ", ")
}

fn texts(values: []json.Value) -> []str {
    var out: []str = []
    for v in values {
        push(mut out, json.stringify(v, 0))
    }
    out
}

fn query(catalog: json.Value, source: str) {
    let nodes = path.find(catalog, source) catch {
        Syntax(at, message) => {
            print("{source}: {message} at {at}")
            return
        }
    }
    if nodes.len == 0 {
        print("{source} -> no match")
        return
    }
    print("{source} -> {show(nodes)}")
}

/// Runs JSONPath queries over a catalog: names, filters, slices and descendants, then lists where the matches live.
fn main() {
    let catalog = parse("{{'books': [{{'title': 'Dune', 'price': 9.5, 'tags': ['scifi', 'classic']}}, {{'title': 'Emma', 'price': 4, 'tags': ['classic']}}, {{'title': 'Neuromancer', 'price': 12, 'tags': ['scifi']}}], 'shop': {{'name': 'Corner Books', 'open': true}}}}")

    query(catalog, "$.shop.name")
    query(catalog, "$.books[*].title")
    query(catalog, "$.books[?@.price < 10].title")
    query(catalog, "$.books[?@.tags[?@ == 'scifi']].title")
    query(catalog, "$.books[-1:]")
    query(catalog, "$..price")
    query(catalog, "$.books[5]")
    query(catalog, "$.books[?@.price <]")

    let cheap = path.compile("$.books[?@.price < 10]") or return
    let nodes = path.query(catalog, cheap)
    print(text.join(path.paths(nodes), "\n"))
}

Functions

compile

pub fn compile(source: str) -> Path ! Error

Compiles the JSONPath query source (RFC 9535).

The query starts with $ and holds segments: .name, .*, .. for descendants, and bracketed selectors that are names, *, indexes, slices start:end:step and filters ?expr. A filter takes comparisons, existence tests, &&, ||, !, nested queries starting with @ or $, and the functions length, count, match, search and value. Fails with Syntax for anything else, including a malformed filter, wrong function arguments, a leading zero in an integer or a bare literal used as a test.

find

pub fn find(v: json.Value, source: str) -> []Node ! Error

Compiles source and returns the nodes it selects from v.

Fails with Syntax when the query is invalid. See compile and query.

paths

pub fn paths(nodes: []Node) -> []str

Returns the normalised paths of nodes, in order.

query

pub fn query(v: json.Value, p: Path) -> []Node

Returns the nodes that p selects from v, in the order the query produces them.

Each node carries its normalised path. A node can appear more than once when several selectors or segments reach it, and an empty result means nothing matched. A missing member or an index out of range selects nothing and is not an error.
