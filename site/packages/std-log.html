std/log

#observability

import std/log

Leveled line logging (debug, info, warn, error) to a descriptor (stderr by default) as level prefix: message, without timestamps so journald and other collectors add their own. with_prefix nests a prefix. A Logger also carries attributes (str_attr, int_attr, float_attr, bool_attr, with_attrs, and the _with level forms), a Format of Text (key=value, quoting what needs it) or Json (one object per line), a Sink (a descriptor, an io.Stream, a line closure or a Record handler), optional RFC 3339 timestamps, and a level from parse_level or the LOG_LEVEL variable. Each record is one write, so lines from green threads do not interleave. A failed write is passed to Logger.on_error (set with with_error_handler), which ignores it by default.

Imports std/io, std/json, std/os, std/sys, std/text, std/time, std/time/datetime, std/time/timestamp.

Examples

service

import std/log

fn handle(l: log.Logger, path: str, status: i64, millis: f64) {
    let fields = [
        log.str_attr("path", path),
        log.int_attr("status", status),
        log.float_attr("ms", millis)
    ]
    if status >= 500 {
        l.error_with("request failed", fields)
    } else if status >= 400 {
        l.warn_with("request rejected", fields)
    } else {
        l.info_with("request served", fields)
    }
}

/// Logs a small service to standard output with a prefix per component, bound attributes and a level read from configuration.
fn main() {
    let level = log.parse_level("  Warning ") or log.Level.Info
    print("configured level {level}")

    let root = log.Logger { level: log.Level.Debug }.with_fd(1).with_prefix("shop")
    let http = root.with_prefix("http").with_attrs([log.str_attr("host", "shop.example")])
    let store = root.with_prefix("store")

    root.info("starting")
    store.debug("opening cache")
    handle(http, "/cart", 200, 3.5)
    handle(http, "/cart/9", 404, 0.75)
    handle(http, "/checkout", 502, 1200.0)
    store.warn_with("slow query", [log.str_attr("sql", "select * from orders where id = 9"), log.int_attr("rows", 0)])

    let quiet = http.with_level(level)
    print("quiet logger shows info: {quiet.enabled(log.Level.Info)}, warnings: {quiet.enabled(log.Level.Warn)}")
    handle(quiet, "/health", 200, 0.1)
    handle(quiet, "/admin", 403, 0.2)

    let unknown = log.parse_level("loud") catch {
        Unknown => log.Level.Info
    }
    print("fell back to {unknown}")
}

import std/log
import std/time/clock
import std/time/duration
import std/time/timestamp

struct Counts {
    errors: i64
    warnings: i64
}

/// Emits JSON and timestamped text lines from a manual clock, and counts severities through a record handler.
fn main() {
    let start = timestamp.parse("2026-11-13T09:00:00Z") or timestamp.epoch()
    let manual = clock.manual(start)
    let c = clock.clock(manual)
    let reading = fn() -> i64 {
        timestamp.unix_nanos(c.read()) ?? 0
    }

    let json = log.Logger { format: log.Format.Json }.with_fd(1).with_prefix("billing").with_timestamps(true).with_clock(reading)
    json.info("batch started")
    clock.advance(manual, duration.milliseconds(1250))
    json.warn_with("retrying", [
        log.str_attr("customer", "ada \"the first\""),
        log.int_attr("attempt", 2),
        log.bool_attr("final", false)
    ])
    clock.advance(manual, duration.seconds(30))
    json.error_with("charge failed", [log.float_attr("amount", 19.99)])

    let text = json.with_format(log.Format.Text)
    clock.advance(manual, duration.minutes(5))
    text.info_with("batch finished", [log.int_attr("charged", 41), log.int_attr("failed", 1)])

    let tally = new(Counts { errors: 0, warnings: 0 })
    let counting = log.Logger { level: log.Level.Debug }.with_handler(fn(r: log.Record) {
        match r.level {
            Warn => { tally.warnings += 1 }
            Error => { tally.errors += 1 }
            _ => {}
        }
    })
    counting.debug("ignored")
    counting.warn("a")
    counting.warn("b")
    counting.error("c")
    counting.with_level(log.Level.Error).warn("filtered out")
    print("warnings {tally.warnings}, errors {tally.errors}")

    let none_yet: []str = []
    let captured = new(none_yet)
    let lines = log.Logger {}.with_lines(fn(line: str) { push(mut *captured, line) })
    lines.info_with("captured", [log.int_attr("n", 1)])
    lines.info("again")
    print("captured {(*captured).len}: {*captured}")
}

with_lines

Returns a copy of the logger that passes each formatted line, without a newline, to f, replacing any other sink.

with_prefix

pub fn with_prefix(l: Logger, prefix: str) -> Logger

Returns a copy of the logger that shows prefix before each message. When the logger already has a prefix, the result is the old prefix, . and prefix.

with_stream

pub fn with_stream(l: Logger, s: io.Stream) -> Logger

Returns a copy of the logger that writes lines to the stream s, replacing any other sink.

with_timestamps

pub fn with_timestamps(l: Logger, on: bool) -> Logger

Returns a copy of the logger that stamps records with the time when on is true, and omits it when false. Timestamps are UTC, RFC 3339 with millisecond precision, taken from the logger's clock.
