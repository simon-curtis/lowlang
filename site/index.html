Low

A compiled, garbage-collected language with green threads, enums and typed errors. It builds small static binaries for Linux and Windows.

enum Shape {
    Circle(radius: i64)
    Rect(w: i64, h: i64)
}

fn area(s: Shape) -> i64 {
    match s {
        Circle(r) => 3 * r * r
        Rect(w, h) => w * h
    }
}

fn main() {
    print(area(Shape.Rect(3, 4)))
}

Runtime and standard library

fn worker(jobs: chan[i64], results: chan[i64]) {
    for j in jobs { results <- j * j }
}

fn main() {
    let jobs = chan[i64](16)
    let results = chan[i64](16)
    for _ in 0..4 { spawn worker(jobs, results) }
    for i in 1..11 { jobs <- i }
    close(jobs)
    var total = 0
    for _ in 0..10 { total += (<-results) ?? 0 }
    print(total)
}
